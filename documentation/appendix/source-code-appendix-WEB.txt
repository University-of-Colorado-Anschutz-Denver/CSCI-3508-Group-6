<section class="wrap" id="content">
  <h1>User guide</h1>
  <p class="lead"><a href="download.html">Download QuadM</a>.</p>
  <div class="guide">
    <!-- Jump links stay visible beside the text on wide screens (recognition over recall). -->
    <nav class="toc" aria-label="On this page">
      <strong>On this page</strong>
      <a href="#open">Find a file</a>
      <a href="#view">View metadata</a>
      <a href="#edit">Edit metadata</a>
      <a href="#search">Search via metadata</a>
      <a href="#faq">FAQ</a>
    </nav>
    <article>
      <!-- Explain how to select a folder for scanning. -->
      <h2 id="open">Find a file</h2>
      <h4>Option 1: Manually find the folder</h4>
      <p>
        <ol>
          <li>Select the drive you want to navigate in the left-most panel through the top dropdown.</li>
          <li>The middle panel will show files and folders contained in the drive. Double-click to open a folder.</li>
          <li>Continue navigating through folders until you find your desired file.</li>
        </ol>
      </p>
      <p class="note info"><svg class="icon" aria-hidden="true" viewBox="0 0 24 24"><circle cx="12" cy="12" r="10"/><path d="M12 11v6M12 7.5h.01"/></svg><span>You can designate a default drive to open upon launching the app in the Settings menu.</span></p>
      <h4>Option 2: Enter the file path</h4>
      <p>
        <ol>
          <li>Find or copy the full file path for the file you want to find.</li>
          <li>Enter it in the top search bar and hit Enter.</li>
          <li>Entering a folder name will show all files in the folder. Entering a file name at the end will show only that file.</li>
        </ol>
      </p>

      <!-- Describe viewing metadata info for supported files. -->
      <h2 id="view">View metadata</h2>
      <p>Single-click a file in the current directory in the center panel. The metadata information will be displayed in the right panel, organized into sections by metadata type.</p>

      <!-- Describe adding and removing tags before saving. -->
      <h2 id="edit">Edit metadata</h2>
      <h4>Editing existing metadata fields</h4>
      <p>
        <ol>
          <li>Once you've selected a file, press the <em>Edit Metadata</em> button at the top of the right panel.</li>
          <li>A window will open displaying currently filled fields, as well as common fields for that file type.</li>
          <li>Type in your new metadata information in the desired field, then press <em>Save Changes</em> in the bottom right corner. A notice will be displayed if changes were saved successfully or if an error was detected.</li>
        </ol>
      </p>
      <h4>Adding custom tags</h4>
      <p>
        <ol>
          <li>Once you've selected a file, press the <em>Edit Metadata</em> button at the top of the right panel.</li>
          <li>A window will open displaying currently filled fields, as well as common fields for that file type.</li>
          <li>Press the <em>+ Add Metadata</em> button in the top right corner.</li>
          <li>In the first dialog box, type the name of the custom metadata field you would like to add.
            <ul>
              <li>Not all file types support editing all metadata types. A good rule of thumb is using <em>ID3</em> for audio files and <em>XMP</em> for other file types.</li>
              <li>For example, to add a custom field named "Tag 1" with the value "good take", enter <em>XMP:Tag1</em>, press OK, then enter <em>good take</em> and press OK.</li>
            </ul>
          </li>
          <li>Press <em>Save Changes</em> in the bottom right corner. A notice will be displayed if changes were saved successfully or if an error was detected.</li>
        </ol>
      </p>
      <p class="note info"><svg class="icon" aria-hidden="true" viewBox="0 0 24 24"><circle cx="12" cy="12" r="10"/><path d="M12 11v6M12 7.5h.01"/></svg><span>Custom field names can only contain letters and numbers with no spaces.</span></p>

      <!-- Explain how to search tags within a chosen folder scope. -->
      <h2 id="search">Search via metadata</h2>
      <p>
        <ol>
          <li>Navigate to the folder you would like to search in.</li>
          <li>In the top bar of the center panel, type in the value you would like to search for among all metadata fields.</li>
          <li>When the search is complete, all files in the current folder that have the specified term in any metadata field will be displayed.</li>
        </ol>
      </p>
      <p class="note info"><svg class="icon" aria-hidden="true" viewBox="0 0 24 24"><circle cx="12" cy="12" r="10"/><path d="M12 11v6M12 7.5h.01"/></svg><span>Only files in the current folder, not sub-folders, will be displayed.</span></p>

      <!-- Answer common questions about file changes, indexing, and support. -->
      <h2 id="faq">FAQ</h2>
      <details><summary>Does the app change my original files?</summary><p>Yes, but only when you press Save. The only data chnaged is metadata, so your file will still work the same in other applications.</p></details>
      <details><summary>How is new metadata information stored?</summary><p>Metadata is embedded in all files, and your new tags are stored in the same data chunks as existing metadata.</p></details>
      <details><summary>I've searched for a file in QuadM. How do I use it in another application?</summary><p>Right-click a file to copy its file path. Then, you can find it in your file explorer and use as normal.</p></details>
      <details><summary>Something is wrong. Where do I report it?</summary><p>Use the <a href="feedback.html">feedback form</a>.</p></details>
    </article>

// Validates the feedback form and writes it to Cloud Firestore through the
// public REST API. Shared by the browser and Node (tests). The same rules are
// enforced a second time, server side, by firebase/firestore.rules.
(function (root, factory) {
  if (typeof module === "object" && module.exports) module.exports = factory();
  else root.FeedbackService = factory();
})(typeof self !== "undefined" ? self : this, function () {
  const TYPES = ["bug", "feature", "question", "other"];
  const EMAIL = /^[^@\s]+@[^@\s]+\.[^@\s]+$/;
  const VERSION = /^[0-9A-Za-z.+-]{1,32}$/;

  // Returns { ok, errors: {field: message}, clean }. clean has only schema fields.
  function validate(input) {
    const d = input || {};
    const errors = {};
    const type = String(d.type || "").trim().toLowerCase();
    const message = String(d.message || "").trim();
    const email = String(d.email || "").trim();
    const appVersion = String(d.version || d.appVersion || "").trim();
    if (!TYPES.includes(type)) errors.type = "Choose a type.";
    if (!message) errors.message = "Please write a message.";
    else if (message.length > 2000) errors.message = "Message is longer than 2000 characters.";
    if (email && (email.length > 254 || !EMAIL.test(email))) errors.email = "Email looks invalid.";
    if (appVersion && !VERSION.test(appVersion)) errors.version = "Version looks invalid (example: 0.1.0).";
    const clean = { type, message };
    if (email) clean.email = email;
    if (appVersion) clean.appVersion = appVersion;
    return { ok: Object.keys(errors).length === 0, errors, clean };
  }

  function randomId(cryptoImpl) {
    const c = cryptoImpl || (typeof crypto !== "undefined" ? crypto : null);
    const chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
    const bytes = new Uint8Array(20);
    if (c && c.getRandomValues) c.getRandomValues(bytes);
    else for (let i = 0; i < 20; i++) bytes[i] = Math.floor(Math.random() * 256);
    return Array.from(bytes, b => chars[b % chars.length]).join("");
  }

  // Firestore REST "commit" with a server-side timestamp, so createdAt is never
  // chosen by the browser. currentDocument.exists=false makes it a create.
  function buildCommit(clean, projectId, id) {
    const fields = {
      type: { stringValue: clean.type },
      message: { stringValue: clean.message },
      status: { stringValue: "new" }
    };
    if (clean.email) fields.email = { stringValue: clean.email };
    if (clean.appVersion) fields.appVersion = { stringValue: clean.appVersion };
    const name = `projects/${projectId}/databases/(default)/documents/feedback/${id}`;
    return {
      writes: [{
        update: { name, fields },
        updateMask: { fieldPaths: Object.keys(fields) },
        updateTransforms: [{ fieldPath: "createdAt", setToServerValue: "REQUEST_TIME" }],
        currentDocument: { exists: false }
      }]
    };
  }

  // cfg: { projectId, apiKey }. Resolves {ok:true} or {ok:false, reason}.
  async function submit(input, cfg, fetchFn, idFn) {
    const v = validate(input);
    if (!v.ok) return { ok: false, reason: "invalid", errors: v.errors };
    if (!cfg || !cfg.projectId || !cfg.apiKey) return { ok: false, reason: "not-configured" };
    const f = fetchFn || (typeof fetch !== "undefined" ? fetch : null);
    const url = `https://firestore.googleapis.com/v1/projects/${cfg.projectId}/databases/(default)/documents:commit?key=${encodeURIComponent(cfg.apiKey)}`;
    try {
      const r = await f(url, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(buildCommit(v.clean, cfg.projectId, (idFn || randomId)()))
      });
      return r.ok ? { ok: true } : { ok: false, reason: "http-" + r.status };
    } catch (e) {
      return { ok: false, reason: "network" };
    }
  }

  return { TYPES, validate, buildCommit, submit, randomId };
});

function initFeedbackForm(form) {
  const status = byId("form-status");
  const submitBtn = form.querySelector("button[type=submit]");
  const counter = byId("message-count");
  const FIELDS = ["type", "email", "message", "version"];
  const MAX = 2000;
  const touched = new Set();      // fields the visitor has already left once
  let sent = false;               // true right after a successful send

  const values = () => Object.fromEntries(new FormData(form));

  // Show or clear the inline error under one field.
  function showFieldError(name, message) {
    const input = form.elements[name], err = byId("err-" + name);
    if (message) { err.textContent = message; err.hidden = false; input.setAttribute("aria-invalid", "true"); }
    else { err.textContent = ""; err.hidden = true; input.removeAttribute("aria-invalid"); }
  }
  function validateField(name) {
    const message = FeedbackService.validate(values()).errors[name];
    showFieldError(name, message);
    return !message;
  }
  function updateCounter() {
    const n = form.elements.message.value.length;
    counter.textContent = `${n} / ${MAX}`;
    counter.classList.toggle("warn", n > MAX * 0.9);
  }
  function setBusy(busy) {
    submitBtn.disabled = busy;
    submitBtn.classList.toggle("loading", busy);
    submitBtn.textContent = busy ? "Sending…" : "Send feedback";
    form.setAttribute("aria-busy", String(busy));
  }
  const hasUnsentText = () => !sent && form.elements.message.value.trim().length > 0;

  // Real-time validation: check a field when the visitor leaves it, then keep
  // re-checking as they edit it (so the error disappears as soon as it is fixed).
  for (const name of FIELDS) {
    const el = form.elements[name];
    el.addEventListener("blur", () => { touched.add(name); validateField(name); });
    el.addEventListener("input", () => {
      sent = false;
      if (name === "message") updateCounter();
      if (touched.has(name)) validateField(name);
    });
  }
  updateCounter();

  // Expert shortcut: Ctrl+Enter (or Cmd+Enter) in the message box sends the form.
  form.elements.message.addEventListener("keydown", e => {
    if ((e.ctrlKey || e.metaKey) && e.key === "Enter") { e.preventDefault(); form.requestSubmit(); }
  });

  // Warn before leaving the page with a message that was not sent.
  window.addEventListener("beforeunload", e => {
    if (hasUnsentText()) { e.preventDefault(); e.returnValue = ""; }
  });

  form.addEventListener("submit", async e => {
    e.preventDefault();                              // handle the submit ourselves via fetch
    const data = values();
    if (data.website) return;                        // honeypot: bots fill this
    clearStatus(status);

    // Check every field first and point at the first problem.
    const errors = FeedbackService.validate(data).errors;
    FIELDS.forEach(name => { touched.add(name); showFieldError(name, errors[name]); });
    const firstBad = FIELDS.find(name => errors[name]);
    if (firstBad) {
      setStatus(status, "err", "Please fix the highlighted field" + (Object.keys(errors).length > 1 ? "s" : "") + " and try again.");
      form.elements[firstBad].focus();
      return;
    }

    setBusy(true);
    setStatus(status, "info", "Sending your feedback…");
    const res = await FeedbackService.submit(data, FIREBASE);
    setBusy(false);

    if (res.ok) {
      form.reset(); touched.clear(); sent = true; updateCounter();
      FIELDS.forEach(name => showFieldError(name, ""));
      setStatus(status, "ok", "Thanks! Your feedback was sent.");
    } else if (res.reason === "invalid") {
      setStatus(status, "err", Object.values(res.errors)[0]);
    } else if (res.reason === "not-configured") {
      setStatus(status, "warn", "Feedback isn't available right now. Please try again later, or contact a team member from the About page.");
    } else if (res.reason === "network") {
      setStatus(status, "err", "We couldn't reach the server. Check your internet connection and try again. Your message is still here.");
    } else {
      setStatus(status, "err", "Something went wrong on our side (" + res.reason + "). Your message is still here, so you can try again in a moment.");
    }
  });
}

// Shared by the browser (download page) and Node (scripts/fetch-release.js).
// Turns a GitHub "latest release" API response into the small object the
// website needs. One source of truth, so the page and the CI step agree.
//
// Files it looks for in a release:
//   .msi  Windows installer   (falls back to a Setup .exe until the pipeline builds an .msi)
//   .deb  Ubuntu/Debian installer, amd64
//   .pkg  macOS graphical installer      (Apple Silicon, and Intel if its name says x86_64 / intel)
//   .dmg  macOS non-graphical installer  (same rule)
(function (root, factory) {
  if (typeof module === "object" && module.exports) module.exports = factory();
  else root.ReleaseData = factory();
})(typeof self !== "undefined" ? self : this, function () {
  function asset(a) {
    return a ? { name: a.name, url: a.browser_download_url, size: a.size || 0 } : null;
  }
  const ext = (a, e) => a.name.toLowerCase().endsWith(e);

  function buildLatest(release, repo) {
    if (!release || !release.tag_name) return null;
    const assets = (release.assets || []).filter(a => a && a.name && a.browser_download_url);

    const windows = assets.find(a => ext(a, ".msi"))
      || (() => {
        const exes = assets.filter(a => ext(a, ".exe"));
        return exes.find(a => /setup|install/i.test(a.name)) || exes[0];
      })()
      || null;

    // amd64 only: skip arm64 builds, and prefer a name that says amd64 / x86_64.
    const debs = assets.filter(a => ext(a, ".deb") && !/arm64|aarch64/i.test(a.name));
    const linux = debs.find(a => /amd64|x86[_-]64/i.test(a.name)) || debs[0] || null;

    // Two Mac builds. A name with x86_64 / intel / amd64 is the Intel build; anything else
    // (arm64, or no architecture in the name) is Apple Silicon.
    const isIntel = a => /x86[_-]?64|intel|amd64/i.test(a.name);
    const macPkgs = assets.filter(a => ext(a, ".pkg"));
    const macDmgs = assets.filter(a => ext(a, ".dmg"));
    const macPkg = macPkgs.find(a => !isIntel(a)) || null;
    const macDmg = macDmgs.find(a => !isIntel(a)) || null;
    const macIntelPkg = macPkgs.find(isIntel) || null;
    const macIntelDmg = macDmgs.find(isIntel) || null;
    const checksums = assets.find(a => /sha-?256/i.test(a.name)) || null;

    return {
      repo: repo || null,
      tag: release.tag_name,
      version: String(release.tag_name).replace(/^v/i, ""),
      publishedAt: release.published_at || null,
      htmlUrl: release.html_url || null,
      assets: {
        windows: asset(windows),
        linux: asset(linux),
        macPkg: asset(macPkg),
        macDmg: asset(macDmg),
        macIntelPkg: asset(macIntelPkg),
        macIntelDmg: asset(macIntelDmg),
        checksums: asset(checksums)
      }
    };
  }

  return { buildLatest };
});

#!/usr/bin/env node
// Writes data/latest.json from the newest published GitHub Release.
// CI runs this right before the Docker image is built, so every image ships
// with the current download links baked in (no browser-side rate limits).
//
//   GITHUB_REPOSITORY=owner/repo GITHUB_TOKEN=... node scripts/fetch-release.js
//
// A missing release is not an error: the file is written with "tag": null and
// the download page falls back to asking the GitHub API itself.
const fs = require("node:fs");
const path = require("node:path");
const { buildLatest } = require("../js/release-data.js");

async function run({ repo, token, fetchFn = fetch, outPath, fsImpl = fs, now = () => new Date() }) {
  if (!repo) throw new Error("GITHUB_REPOSITORY is not set (expected owner/name)");
  const headers = { Accept: "application/vnd.github+json", "User-Agent": "quadm-website-build" };
  if (token) headers.Authorization = `Bearer ${token}`;

  let latest = null;
  let note = "";
  try {
    const res = await fetchFn(`https://api.github.com/repos/${repo}/releases/latest`, { headers });
    if (res.status === 404) note = "no published release yet";
    else if (!res.ok) note = `GitHub answered ${res.status}`;
    else latest = buildLatest(await res.json(), repo);
  } catch (err) {
    note = `lookup failed: ${err.message}`;
  }

  const out = latest || { repo, tag: null, version: null, assets: {} };
  out.generatedAt = now().toISOString();
  fsImpl.mkdirSync(path.dirname(outPath), { recursive: true });
  fsImpl.writeFileSync(outPath, JSON.stringify(out, null, 2) + "\n");
  return { latest, note, outPath };
}

module.exports = { run };

if (require.main === module) {
  run({
    repo: process.env.GITHUB_REPOSITORY,
    token: process.env.GITHUB_TOKEN,
    outPath: process.env.OUT || path.join(__dirname, "..", "data", "latest.json")
  }).then(({ latest, note, outPath }) => {
    if (latest) console.log(`Wrote ${outPath}: ${latest.tag}`);
    else console.log(`::warning::${note}; wrote ${outPath} without a release (page will ask GitHub directly)`);
  }).catch(err => { console.error(err.message); process.exit(1); });
}
