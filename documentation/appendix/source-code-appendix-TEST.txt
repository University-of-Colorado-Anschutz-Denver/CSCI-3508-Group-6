// Closes the next QMessageBox the app opens (they block until answered) and remembers its text.
class DialogCatcher : public QObject
{
public:
    DialogCatcher()
    {
        connect(&timer, &QTimer::timeout, this, [this]() {
            if (auto *box = qobject_cast<QMessageBox *>(QApplication::activeModalWidget())) {
                text = box->text();
                seen = true;
                box->close();
                timer.stop();
            }
        });
        timer.start(30);
    }
    QTimer timer;
    QString text;
    bool seen = false;
};

QString clean(const QString &path) { return QDir::cleanPath(QDir::fromNativeSeparators(path)); }

QListWidgetItem *itemByText(QListWidget *list, const QString &text)
{
    for (int i = 0; i < list->count(); ++i) {
        if (list->item(i)->text() == text)
            return list->item(i);
    }
    return nullptr;
}

QStringList itemTexts(QListWidget *list, bool visibleOnly = false)
{
    QStringList texts;
    for (int i = 0; i < list->count(); ++i) {
        if (!visibleOnly || !list->item(i)->isHidden())
            texts << list->item(i)->text();
    }
    return texts;
}

QPoint centerOf(QListWidget *list, QListWidgetItem *item)
{
    list->scrollToItem(item);
    return list->visualItemRect(item).center();
}

void clickItem(QListWidget *list, QListWidgetItem *item)
{
    QTest::mouseClick(list->viewport(), Qt::LeftButton, Qt::NoModifier, centerOf(list, item));
}

void doubleClickItem(QListWidget *list, QListWidgetItem *item)
{
    // Qt's synthetic double click needs a preceding normal click on the same item
    clickItem(list, item);
    QTest::mouseDClick(list->viewport(), Qt::LeftButton, Qt::NoModifier, centerOf(list, item));
}

bool exifToolAvailable() { return !ExifTool::executablePath().isEmpty(); }

class MainWindowTests : public QObject
{
    Q_OBJECT

private:
    QTemporaryDir data;  // folder full of test files
    QString dataPath;
    MainWindow *window = nullptr;

    QLineEdit *pathBar() { return window->findChild<QLineEdit *>("filePathSearchBar"); }
    QLineEdit *tagBar() { return window->findChild<QLineEdit *>("tagFilterBar"); }
    QListWidget *contents() { return window->findChild<QListWidget *>("contentsListWidget"); }
    QListWidget *roots() { return window->findChild<QListWidget *>("rootListWidget"); }
    QLabel *pathLabel() { return window->findChild<QLabel *>("currentPathLabel"); }
    QLabel *metadataLabel() { return window->findChild<QLabel *>("metadataPlaceholderLabel"); }

    QPushButton *expandButton() { return window->findChild<QPushButton *>("expandMetadataButton"); }

    void goTo(const QString &path)
    {
        pathBar()->setText(path);
        QTest::keyClick(pathBar(), Qt::Key_Return);
    }

    {
        QImage image(8, 8, QImage::Format_RGB32);
        image.fill(Qt::red);
        QVERIFY(image.save(path));

    void initTestCase()
    {
        QVERIFY(data.isValid());
        dataPath = data.path();
        QDir dir(dataPath);
        QVERIFY(dir.mkdir("subfolder"));
        makeImage(dir.filePath("photo_alpha.png"));
        makeImage(dir.filePath("photo_beta.png"));
        makeImage(dir.filePath("subfolder/inner.png"));

        QFile notes(dir.filePath("notes.txt"));  // not media, must not be listed
        QVERIFY(notes.open(QIODevice::WriteOnly));
        notes.write("hello");
    }

    void init()
    {
        window = new MainWindow;
        window->show();
        QVERIFY(QTest::qWaitForWindowExposed(window));
    }

    {
        delete window;
        window = nullptr;
    }

        open->trigger();

                dialog = s;

class MetadataEditorTests : public QObject
{
    Q_OBJECT

private:
    QTemporaryDir data;
    QString imagePath;

    QString newImage(const QString &name)
    {
        const QString path = data.filePath(name);
        QImage image(8, 8, QImage::Format_RGB32);
        image.fill(Qt::blue);
        image.save(path);
        return path;
    }

    static QLineEdit *fieldWithText(QWidget *editor, const QString &text)
    {
        for (QLineEdit *edit : editor->findChildren<QLineEdit *>()) {
            if (edit->text() == text)
                return edit;
        }
        return nullptr;
    }

private slots:
    void initTestCase() { QVERIFY(data.isValid()); }

    void init() { imagePath = newImage("editor_test.png"); }

    // UI-20
    void opens_showsFileNameAndType()
    {
        MetadataEditor editor(imagePath);
        editor.show();
        QVERIFY(QTest::qWaitForWindowExposed(&editor));
        QCOMPARE(editor.findChild<QLabel *>("fileNameLabel")->text(), QString("editor_test.png"));
        QCOMPARE(editor.findChild<QLabel *>("fileTypeLabel")->text(), QString("PNG"));
        QCOMPARE(editor.windowTitle(), QString("QuadM - Metadata Editor"));
    }

    // UI-21
    void missingFile_showsNotFound()
    {
        MetadataEditor editor(data.filePath("missing.png"));
        QCOMPARE(editor.findChild<QLabel *>("fileNameLabel")->text(), QString("File not found."));
    }

    // UI-22
    void cancelButton_closesWindow()
    {
        MetadataEditor editor(imagePath);
        editor.show();
        QVERIFY(QTest::qWaitForWindowExposed(&editor));
        QTest::mouseClick(editor.findChild<QPushButton *>("cancelButton"), Qt::LeftButton);
        QVERIFY(!editor.isVisible());
    }

    // UI-23
    void fields_listed_withEditableFileName()
    {
        if (!exifToolAvailable())
            QSKIP("ExifTool is not next to the test executable");
        MetadataEditor editor(imagePath);
        QLineEdit *name = fieldWithText(&editor, "editor_test.png");
        QVERIFY2(name, "File name field not shown");
        QVERIFY(!name->isReadOnly());
        QVERIFY(editor.findChildren<QLineEdit *>().size() > 1);
    }

    // UI-24
    void readOnlyFields_haveDisabledDeleteButton()
    {
        if (!exifToolAvailable())
            QSKIP("ExifTool is not next to the test executable");
        MetadataEditor editor(imagePath);
        bool foundReadOnly = false;
        for (QLineEdit *edit : editor.findChildren<QLineEdit *>()) {
            if (!edit->isReadOnly())
                continue;
            foundReadOnly = true;
            auto *delButton = edit->parentWidget()->findChild<QPushButton *>();
            QVERIFY(delButton);
            QVERIFY(!delButton->isEnabled());
        }
        QVERIFY2(foundReadOnly, "Expected at least one read-only field (e.g. image size)");
    }

    // UI-25
    void deleteButton_removesRowFromEditor()
    {
        if (!exifToolAvailable())
            QSKIP("ExifTool is not next to the test executable");
        MetadataEditor editor(imagePath);
        editor.show();
        QVERIFY(QTest::qWaitForWindowExposed(&editor));

        QPushButton *target = nullptr;
        for (QPushButton *b : editor.findChildren<QPushButton *>()) {
            if (b->text() == "Delete" && b->isEnabled())
                target = b;
        }
        if (!target)
            QSKIP("No deletable field in this file");
        QWidget *row = target->parentWidget();
        const int before = editor.findChildren<QLineEdit *>().size();
        QTest::mouseClick(target, Qt::LeftButton);
        QTRY_COMPARE(editor.findChildren<QLineEdit *>().size(), before - 1);
        Q_UNUSED(row);
    }

    // UI-26
    void save_withoutChanges_saysNoChanges()
    {
        if (!exifToolAvailable())
            QSKIP("ExifTool is not next to the test executable");
        MetadataEditor editor(imagePath);
        editor.show();
        QVERIFY(QTest::qWaitForWindowExposed(&editor));
        QSignalSpy saved(&editor, &MetadataEditor::metadataSaved);

        DialogCatcher dialog;
        QTest::mouseClick(editor.findChild<QPushButton *>("saveMetadataButton"), Qt::LeftButton);
        QVERIFY(dialog.seen);
        QCOMPARE(dialog.text, QString("No changes to save."));
        QCOMPARE(saved.count(), 0);
    }

    // UI-27
    void save_renamedFile_renamesOnDisk()
    {
        if (!exifToolAvailable())
            QSKIP("ExifTool is not next to the test executable");
        MetadataEditor editor(imagePath);
        editor.show();
        QVERIFY(QTest::qWaitForWindowExposed(&editor));
        QSignalSpy saved(&editor, &MetadataEditor::metadataSaved);

        QLineEdit *name = fieldWithText(&editor, "editor_test.png");
        QVERIFY(name);
        name->setText("renamed.png");

        DialogCatcher dialog;
        QTest::mouseClick(editor.findChild<QPushButton *>("saveMetadataButton"), Qt::LeftButton);
        QVERIFY(dialog.seen);
        QVERIFY(dialog.text.contains("saved successfully"));
        QCOMPARE(saved.count(), 1);
        QVERIFY(QFile::exists(data.filePath("renamed.png")));
        QVERIFY(!QFile::exists(imagePath));
    }

    // UI-28
    void save_renameToExistingName_isRejected()
    {
        if (!exifToolAvailable())
            QSKIP("ExifTool is not next to the test executable");
        const QString other = newImage("already_taken.png");
        MetadataEditor editor(imagePath);
        editor.show();
        QVERIFY(QTest::qWaitForWindowExposed(&editor));
        QSignalSpy saved(&editor, &MetadataEditor::metadataSaved);

        QLineEdit *name = fieldWithText(&editor, "editor_test.png");
        QVERIFY(name);
        name->setText("already_taken.png");

        DialogCatcher dialog;
        QTest::mouseClick(editor.findChild<QPushButton *>("saveMetadataButton"), Qt::LeftButton);
        QVERIFY(dialog.seen);
        QVERIFY(dialog.text.contains("already exists"));
        QCOMPARE(saved.count(), 0);
        QVERIFY(QFile::exists(imagePath));
        QVERIFY(QFile::exists(other));
    }
};

// Runs one test class; its results go to stdout and to <exe folder>/ui-test-results/<name>.txt
template <typename T>
int runClass(const char *name, int argc, char *argv[])
{
    const QString dir = QCoreApplication::applicationDirPath() + "/ui-test-results";
    QDir().mkpath(dir);
    const QByteArray file = (dir + "/" + name + ".txt,txt").toLocal8Bit();

    QList<QByteArray> args;
    for (int i = 0; i < argc; ++i)
        args << argv[i];
    args << "-o" << file << "-o" << "-,txt";
    QList<char *> ptrs;
    for (QByteArray &a : args)
        ptrs << a.data();

    T test;
    return QTest::qExec(&test, ptrs.size(), ptrs.data());
}

int main(int argc, char *argv[])
{
    QApplication app(argc, argv);

    // Keep the tests away from the real user settings (registry / ~/.config)
    QTemporaryDir settingsDir;
    QSettings::setDefaultFormat(QSettings::IniFormat);
    QSettings::setPath(QSettings::IniFormat, QSettings::UserScope, settingsDir.path());

    int failures = 0;
    failures += runClass<MainWindowTests>("MainWindowTests", argc, argv);
    failures += runClass<MetadataEditorTests>("MetadataEditorTests", argc, argv);
    failures += runClass<SettingsTests>("SettingsTests", argc, argv);
    return failures;
}

name: Release

# Push a version tag (git tag v0.4.0 && git push origin v0.4.0) and this workflow:
#   1. builds the installers with the scripts in code/packaging/, in parallel:
#        Windows  .msi        (build-windows-msi.ps1)
#        Ubuntu   .deb        (build-deb.sh)
#        macOS    .pkg, .dmg  (build-macos.sh), once for Apple Silicon and once for Intel
#   2. installs each one on a clean runner and checks the app + ExifTool start
#   3. publishes the files to a GitHub Release, with SHA256SUMS.txt
#   4. starts website.yml so the website image shows the new download links
# Running it by hand (Actions tab -> Run workflow) builds and tests, nothing is published.
#
# The packaging scripts are the ones in code/packaging/ - the same commands a
# developer runs locally - so CI and a laptop produce the same installers.

on:
  push:
    tags: ["v[0-9]+.[0-9]+.[0-9]+"]
  workflow_dispatch:

permissions:
  contents: read

env:
  QT_VERSION: "6.8.2"        # Qt used on the runners (the presets on a laptop may use another version)
  EXIFTOOL_VERSION: "13.60"  # Linux and macOS: build-deb.sh / build-macos.sh download this exact version (Windows uses the newest, see below)

jobs:
  windows:
    name: Windows .msi
    runs-on: windows-latest
    outputs:
      version: ${{ steps.v.outputs.version }}
    defaults:
      run:
        shell: bash
    steps:
      - uses: actions/checkout@v4

      - name: Version from the tag (0.0.0 for a manual run)
        id: v
        run: |
          if [[ "$GITHUB_REF" == refs/tags/v* ]]; then v="${GITHUB_REF_NAME#v}"; else v="0.0.0"; fi
          echo "version=$v" >> "$GITHUB_OUTPUT"

      # install-qt-action runs aqtinstall with whatever Python the runner has; the runner's newest
      # (3.15) made it fail with exit code 1, so pin a Python that aqtinstall supports.
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      # The script is written for the Qt MinGW kit (same as CMakePresets.json), so install that kit.
      - name: Install Qt (MinGW kit) and the MinGW compiler
        uses: jurplel/install-qt-action@v4
        with:
          version: ${{ env.QT_VERSION }}
          host: windows
          target: desktop
          arch: win64_mingw
          tools: tools_mingw1310
          cache: true

      # build-windows-msi.ps1 refuses to run without code/third_party/exiftool/exiftool.exe.
      # Both public hosts refuse GitHub's runners (SourceForge answers 403, exiftool.org has no direct
      # file), so the Windows zip comes from our own repo: first the copy committed in
      # code/packaging/windows-deps/, then a pre-release tagged "tools-exiftool" (see .github/CICD.md).
      # The public hosts are only a last resort.
      - name: Get the Windows ExifTool
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          dest=code/third_party/exiftool
          ok=0
          mkdir -p "$RUNNER_TEMP/dl"
          committed=$(ls code/packaging/windows-deps/exiftool-*_64.zip 2>/dev/null | sort | tail -n1 || true)
          if [ -n "$committed" ] && unzip -tq "$committed" >/dev/null 2>&1; then
            echo "Using the committed copy: $committed"
            cp "$committed" "$RUNNER_TEMP/exiftool.zip"; ok=1
          fi
          if [ "$ok" != 1 ] && gh release download tools-exiftool --repo "$GITHUB_REPOSITORY" --pattern 'exiftool-*_64.zip' --dir "$RUNNER_TEMP/dl" --clobber 2>/dev/null; then
            f=$(ls "$RUNNER_TEMP"/dl/exiftool-*_64.zip | head -n1)
            echo "Using our copy: $(basename "$f")"
            if unzip -tq "$f" >/dev/null 2>&1; then cp "$f" "$RUNNER_TEMP/exiftool.zip"; ok=1; fi
          fi
          if [ "$ok" != 1 ]; then
            zip=$(curl -fsSL https://exiftool.org/ | grep -oE 'exiftool-[0-9]+\.[0-9]+_64\.zip' | head -n1 || true)
            [ -n "$zip" ] || zip="exiftool-$(curl -fsSL https://exiftool.org/ver.txt | tr -d '[:space:]')_64.zip"
            echo "No usable tools-exiftool release; trying the public hosts for $zip"
            for url in "https://downloads.sourceforge.net/project/exiftool/$zip" "https://sourceforge.net/projects/exiftool/files/$zip/download" "https://exiftool.org/$zip"; do
              echo "Trying $url"
              if curl -fsSL --retry 2 -A "Mozilla/5.0" -o "$RUNNER_TEMP/exiftool.zip" "$url" && unzip -tq "$RUNNER_TEMP/exiftool.zip" >/dev/null 2>&1; then ok=1; break; fi
            done
          fi
          [ "$ok" = 1 ] || { echo "::error::No ExifTool zip. Commit exiftool-XX.XX_64.zip to code/packaging/windows-deps/ (see .github/CICD.md)."; exit 1; }
          unzip -q "$RUNNER_TEMP/exiftool.zip" -d "$RUNNER_TEMP/exiftool"
          src=$(find "$RUNNER_TEMP/exiftool" -maxdepth 1 -mindepth 1 -type d | head -n1)
          cp -R "$src"/. "$dest"/
          [ -f "$dest/exiftool(-k).exe" ] && mv "$dest/exiftool(-k).exe" "$dest/exiftool.exe"
          test -f "$dest/exiftool.exe" && test -d "$dest/exiftool_files"
          "$dest/exiftool.exe" -ver

      - name: Build the MSI
        shell: pwsh
        working-directory: code
        run: |
          $qtRoot = Split-Path -Parent (Split-Path -Parent $env:QT_ROOT_DIR)   # ...\Qt
          $mingw = Join-Path $qtRoot "Tools\mingw1310_64"
          if (-not (Test-Path "$mingw\bin\g++.exe")) { throw "MinGW not found at $mingw" }
          $ninja = Split-Path -Parent (Get-Command ninja).Source
          .\packaging\build-windows-msi.ps1 -Version "${{ steps.v.outputs.version }}" `
            -QtDir $env:QT_ROOT_DIR -MinGwDir $mingw -NinjaDir $ninja

      - name: Install the MSI on this runner and check it
        shell: pwsh
        run: |
          $msi = (Get-ChildItem code\dist\*.msi).FullName
          $p = Start-Process msiexec.exe -ArgumentList "/i `"$msi`" /qn /l*v $env:RUNNER_TEMP\install.log" -Wait -PassThru
          if ($p.ExitCode -ne 0) { Get-Content $env:RUNNER_TEMP\install.log -Tail 60; throw "msiexec exit code $($p.ExitCode)" }
          $dir = "$env:ProgramFiles\QuadM"
          foreach ($f in "QuadM.exe", "exiftool\exiftool.exe", "Qt6Widgets.dll") {
            if (-not (Test-Path "$dir\$f")) { throw "Missing after install: $dir\$f" }
          }
          & "$dir\exiftool\exiftool.exe" -ver
          $app = Start-Process "$dir\QuadM.exe" -PassThru
          Start-Sleep -Seconds 8
          if ($app.HasExited) { throw "QuadM exited right after start (code $($app.ExitCode))" }
          Stop-Process $app
          $u = Start-Process msiexec.exe -ArgumentList "/x `"$msi`" /qn" -Wait -PassThru
          if ($u.ExitCode -ne 0) { throw "uninstall exit code $($u.ExitCode)" }

      - uses: actions/upload-artifact@v4
        with:
          name: windows-msi
          path: code/dist/*.msi
          if-no-files-found: error

# Display name in the GitHub Actions interface.
name: Website checks

# On every change to code/website:
#   pull request -> lint + tests + image build and smoke test (nothing is published)
#   push to a listed branch -> same, then the image is pushed to GitHub Container Registry
#   run manually, or by release.yml after a new app release -> rebuild the
#   image so its download page carries the new installer links
#
# Image: ghcr.io/<owner>/quadm-website   (tags: branch name, short commit hash, latest on main)

# Events that trigger this workflow.
on:
  # Trigger on pushes to the testing branch.
  push:
    # Branches included in the push trigger.
    branches: [main, testing, website_skeleton]
    # Run only when website files or this workflow file change.
    paths: ["code/website/**", ".github/workflows/website.yml"]
  # Trigger on pull requests that modify matching paths.
  pull_request:
    # Files included in the pull-request trigger.
    paths: ["code/website/**", ".github/workflows/website.yml"]
  # Enable manual runs from the GitHub Actions interface (release.yml also starts it this way).
  workflow_dispatch:
    inputs:
      reason:
        description: "Why this run (shown in the log)"
        required: false
        default: "manual run"

# Permissions granted to the workflow's GitHub token.
permissions:
  # Permit read-only access to repository files.
  contents: read

concurrency:
  group: website-${{ github.ref }}
  cancel-in-progress: true

jobs:

    name: Lint and test website

    runs-on: ubuntu-latest

    defaults:
      # Settings applied to run steps.

        # Execute commands from the website project directory.
        working-directory: code/website

      # Check out the repository contents.

      - uses: actions/setup-node@v4
        # Configuration for the Node.js setup action.

      - name: Lint HTML

        run: npm run lint:html

      - name: Run tests
        # Invoke the test script from package.json.

  image:
    name: Build, smoke-test and publish image
    needs: test
    runs-on: ubuntu-latest
    permissions:
      contents: read
      packages: write # push to ghcr.io with the built-in token
    # Publish only from the main branch; other runs build and test without pushing.
    env:
      PUBLISH: ${{ github.event_name != 'pull_request' && contains(fromJSON('["main","website_skeleton","testing"]'), github.ref_name) }}
    steps:
      # Check out the source needed to build the website image.
      - uses: actions/checkout@v4

      # Set up Node.js for any Node-based Docker build steps.
      - uses: actions/setup-node@v4
        with:
          node-version: 20

      # Create a lowercase GHCR image name from the repository owner.
      - name: Name the image (registry names must be lowercase)
        id: name
        run: echo "image=ghcr.io/${GITHUB_REPOSITORY_OWNER,,}/quadm-website" >> "$GITHUB_OUTPUT"

      # Bakes the newest release's download links into the site. A missing
      # release is fine; the page then asks the GitHub API itself.
      - name: Write data/latest.json from the newest release
        working-directory: code/website
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: node scripts/fetch-release.js

      # Enable BuildKit features used by the Docker build actions.
      - uses: docker/setup-buildx-action@v3

      # Generate consistent image tags and OCI metadata labels.
      - name: Image tags and labels
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ${{ steps.name.outputs.image }}
          tags: |
            type=raw,value=latest,enable={{is_default_branch}}

            type=sha,format=short

        with:
          context: code/website
          load: true
          tags: quadm-website:ci
          labels: ${{ steps.meta.outputs.labels }}
          cache-from: type=gha
          cache-to: type=gha,mode=max

      # Start the image and verify expected pages/assets work and build files stay private.
      - name: Smoke-test the container
        run: |
          docker run -d --name site -p 8080:8080 quadm-website:ci
          for i in $(seq 1 30); do curl -fs http://localhost:8080/healthz >/dev/null && break; sleep 1; done
          for p in / /download.html /guide.html /feedback.html /about.html /css/style.css /js/main.js; do
            code=$(curl -s -o /dev/null -w '%{http_code}' "http://localhost:8080$p")
            echo "$p -> $code"; [ "$code" = 200 ] || { docker logs site; exit 1; }
          done
          # release data must be valid JSON
          curl -fs http://localhost:8080/data/latest.json | node -e 'const d=JSON.parse(require("fs").readFileSync(0,"utf8")); console.log("latest.json:", d.tag ?? "no release yet")'
          # test and build files must not be served
          for p in /package.json /tests/site.test.js /scripts/fetch-release.js /Dockerfile /nginx.conf /security-headers.inc; do
            code=$(curl -s -o /dev/null -w '%{http_code}' "http://localhost:8080$p")

          done
          docker rm -f site

      - name: Log in to GitHub Container Registry

          registry: ghcr.io

        uses: docker/build-push-action@v6
        with:
          context: code/website
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          cache-from: type=gha

      # Record whether the image was published and show a command to run it.
      - name: Summary
        if: always()
        run: |
          {
            echo "### Website image"
            echo "Trigger: ${{ github.event_name }} ${{ github.event.inputs.reason }}"
            if [ "$PUBLISH" = true ]; then
              echo ""
              echo "Published: \`${{ steps.name.outputs.image }}:latest\`"
              echo ""
              echo '```'
              echo "docker run --rm -p 8080:8080 ${{ steps.name.outputs.image }}:latest"
              echo '```'
            else
              echo "Built and tested only (not published from this ref)."
            fi
          } >> "$GITHUB_STEP_SUMMARY"
