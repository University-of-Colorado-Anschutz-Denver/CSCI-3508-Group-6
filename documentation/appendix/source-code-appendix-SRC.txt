#pragma once

#include <QJsonObject>
#include <QMap>
#include <QSet>
#include <QString>
#include <QStringList>

#include <optional>

namespace ExifTool {

    // Full path to the bundled ExifTool executable, or an empty string if it isn't found.
    QString executablePath();

    // Reads all metadata of `file` (grouped tag names, e.g. "EXIF:Make") into `out`.
    // On failure returns false and puts a readable message in `error`.
    bool readMetadata(const QString& file, QJsonObject& out, QString& error);

    // File extensions (lower case, e.g. "png") that ExifTool can write. Asked from ExifTool once, then cached.
    const QSet<QString>& writableExtensions();

    // Writes `tags` ("Group:Tag" -> value) into `file`. An empty value removes the tag.
    // If `keywords` is set, the file's keyword list (our tags) is replaced by it.
    // Any problem ExifTool reports (unknown or unwritable tag, ...) makes this return false;
    // tags that were fine may already have been written in that case.
    bool writeTags(const QString& file, const QMap<QString, QString>& tags,
        const std::optional<QStringList>& keywords, QString& error);
    bool writeTags(const QString& file, const QMap<QString, QString>& tags, QString& error);

    // Finds every file in `folder` (and its subfolders if `recursive`) that has tags and
    // returns "file path" -> its tags. Files without tags are left out. Can take a while for big folders.
    bool scanTags(const QString& folder, bool recursive, QMap<QString, QStringList>& out, QString& error);

    // Reads every supported file's metadata in `folder` (and subfolders if `recursive`) in one
    // ExifTool call, keyed by file path. Used to search/filter a folder's files by metadata.
    bool readFolderMetadata(const QString& folder, bool recursive, QMap<QString, QJsonObject>& out,
        QString& error);

} // namespace ExifTool

#include "exiftool.h"

#include "backend/metadata.h"

#include <QCoreApplication>
#include <QFileInfo>
#include <QJsonArray>
#include <QJsonDocument>
#include <QProcess>
#include <QRegularExpression>
#include <QSet>
#include <QStringList>
#include <QTemporaryFile>

namespace ExifTool {

    QString executablePath()
    {
#ifdef Q_OS_WIN
        const QString name = QStringLiteral("exiftool.exe");
#else
        const QString name = QStringLiteral("exiftool");
#endif
        const QString appDir = QCoreApplication::applicationDirPath();

        // Same layout for the build folder and the installed app; the second entry is the macOS .app bundle
        const QStringList candidates = {
            appDir + "/exiftool/" + name,
            appDir + "/../Resources/exiftool/" + name,
        };
        for (const QString& candidate : candidates) {
            if (QFileInfo::exists(candidate))
                return QFileInfo(candidate).absoluteFilePath();
        }
        return {};
    }

    // Spawns "exiftool -json -G -- file" and parses its one-object JSON array into `out`.
    // "-G" prefixes each key with its group (e.g. "EXIF:Make"), which is what lets the rest of
    // the app tell apart same-named tags from different formats.
    bool readMetadata(const QString& file, QJsonObject& out, QString& error)
    {
        const QString exe = executablePath();
        if (exe.isEmpty()) {
            error = QStringLiteral("ExifTool was not found next to the application.");
            return false;
        }

        QProcess process;
        process.start(exe, { "-json", "-G", "--", file });
        if (!process.waitForFinished(15000)) {
            process.kill();
            error = QStringLiteral("ExifTool did not finish: ") + process.errorString();
            return false;
        }
        if (process.exitStatus() != QProcess::NormalExit || process.exitCode() != 0) {
            error = QString::fromUtf8(process.readAllStandardError()).trimmed();
            return false;
        }

        QJsonParseError parseError;
        const QJsonDocument doc = QJsonDocument::fromJson(process.readAllStandardOutput(), &parseError);
        if (parseError.error != QJsonParseError::NoError || !doc.isArray() || doc.array().isEmpty()) {
            error = QStringLiteral("Could not read ExifTool output.");
            return false;
        }
        // ExifTool always returns a JSON array, one object per input file; we only ever pass one file.
        out = doc.array().first().toObject();
        return true;
    }

    // Runs "exiftool -listwf" once (lazily, via the static initializer) and caches the result for
    // the rest of the program's life, since the writable-extensions list never changes at runtime.
    const QSet<QString>& writableExtensions()
    {
        static const QSet<QString> extensions = [] {
            QSet<QString> result;
            const QString exe = executablePath();
            if (exe.isEmpty())
                return result;
            QProcess process;
            process.start(exe, { "-listwf" });
            if (!process.waitForFinished(15000))
                return result;
            // Output: a heading line, then indented rows of upper-case extensions
            const QStringList lines = QString::fromUtf8(process.readAllStandardOutput()).split('\n');
            for (const QString& line : lines) {
                if (!line.startsWith(' '))
                    continue;
                for (const QString& word : line.split(' ', Qt::SkipEmptyParts))
                    result.insert(word.trimmed().toLower());
            }
            return result;
            }();
        return extensions;
    }

    namespace {

        // Runs ExifTool with the given arguments, one per line in a UTF-8 argument file, so accents and
        // symbols in values and paths survive on Windows. Line breaks inside an argument are not possible.
        // `configFile` (optional) is an ExifTool config; it has to come before every other option.
        bool runWithArgFile(const QStringList& lines, int timeoutMs, QString& standardOutput,
            QString& standardError, QString& error, const QString& configFile = QString())
        {
            const QString exe = executablePath();
            if (exe.isEmpty()) {
                error = QStringLiteral("ExifTool was not found next to the application.");
                return false;
            }

            QTemporaryFile argFile;
            if (!argFile.open()) {
                error = QStringLiteral("Could not create a temporary file.");
                return false;
            }
            for (const QString& line : lines)
                argFile.write(line.toUtf8() + "\n");
            argFile.close();

            // "-@ argfile" tells ExifTool to read its arguments from the file instead of argv
            QStringList arguments;
            if (!configFile.isEmpty())
                arguments << "-config" << configFile;
            arguments << "-charset" << "utf8" << "-charset" << "filename=utf8" << "-@" << argFile.fileName();

            QProcess process;
            process.start(exe, arguments);
            if (!process.waitForFinished(timeoutMs)) {
                process.kill();
                error = QStringLiteral("ExifTool did not finish: ") + process.errorString();
                return false;
            }
            standardOutput = QString::fromUtf8(process.readAllStandardOutput());
            standardError = QString::fromUtf8(process.readAllStandardError()).trimmed();
            return true;
        }

        // ExifTool argument files can't contain literal line breaks, so collapse any into spaces.
        QString oneLine(QString text)
        {
            return text.replace('\r', ' ').replace('\n', ' ');
        }

        // A JSON value that is either one string or a list of strings -> list of strings
        QStringList toStringList(const QJsonValue& value)
        {
            QStringList result;
            if (value.isArray()) {
                for (const QJsonValue& item : value.toArray())
                    result << item.toVariant().toString();
            }
            else if (!value.isUndefined() && !value.isNull()) {
                result << value.toVariant().toString();
            }
            return result;
        }

        // XMP tags ExifTool doesn't know (like "XMP:Hello") can't be written as they are. They are
        // stored instead in QuadM's own XMP namespace, which ExifTool still reads back as "XMP:Hello".
        const QString kCustomXmpGroup = QStringLiteral("XMP-quadm");

        // The keys ExifTool reported as "Tag 'X' is not defined" that we can store that way
        QSet<QString> customXmpKeys(const QString& problems, const QMap<QString, QString>& tags)
        {
            static const QRegularExpression undefinedTag("Tag '([^']+)' is not defined");
            static const QRegularExpression validName("^[A-Za-z][A-Za-z0-9]*$");
            QSet<QString> keys;
            for (auto match = undefinedTag.globalMatch(problems); match.hasNext();) {
                const QString key = match.next().captured(1);
                const bool hasGroup = key.contains(':');
                const QString group = key.section(':', 0, 0);
                const QString name = key.section(':', hasGroup ? 1 : 0);
                if (tags.contains(key) && validName.match(name).hasMatch()
                    && (!hasGroup || group.compare(QLatin1String("XMP"), Qt::CaseInsensitive) == 0))
                    keys.insert(key);
            }
            return keys;
        }

        // An ExifTool config file that declares `names` as writable text tags in the custom namespace
        bool writeCustomXmpConfig(QTemporaryFile& file, const QStringList& names)
        {
            if (!file.open())
                return false;
            QString text = QStringLiteral(
                "%Image::ExifTool::UserDefined = ('Image::ExifTool::XMP::Main' => "
                "{ quadm => { SubDirectory => { TagTable => 'Image::ExifTool::UserDefined::quadm' } } });\n"
                "%Image::ExifTool::UserDefined::quadm = (\n"
                "  GROUPS => { 0 => 'XMP', 1 => '%1', 2 => 'Other' },\n"
                "  NAMESPACE => { 'quadm' => 'https://quadm.app/ns/1.0/' },\n"
                "  WRITABLE => 'string',\n").arg(kCustomXmpGroup);
            for (const QString& name : names)
                text += "  " + name + " => { },\n";
            text += QStringLiteral(");\n1;\n");
            file.write(text.toUtf8());
            file.close();
            return true;
        }

    } // namespace

    // Writes `tags` as "-Group:Tag=value" arguments (an empty value deletes that tag), and
    // optionally replaces the keyword list (our tags) with `keywords`.
    bool writeTags(const QString& file, const QMap<QString, QString>& tags,
        const std::optional<QStringList>& keywords, QString& error)
    {
        // `custom` = keys to store in the QuadM XMP namespace instead of under their own name
        auto buildLines = [&](const QSet<QString>& custom) {
            QStringList lines = { QStringLiteral("-overwrite_original") };
            for (auto it = tags.begin(); it != tags.end(); ++it) {
                const QString name = custom.contains(it.key()) ? kCustomXmpGroup + ":" + it.key().section(':', -1)
                                                               : it.key();
                lines << "-" + name + "=" + oneLine(it.value());
            }
            if (keywords) {
                // "=" replaces the whole list (and with no value empties it); "+=" adds to it.
                // Emptying and adding in one go doesn't work, so the first tag uses "=".
                if (keywords->isEmpty())
                    lines << "-" + TagStorage::XmpWriteTag + "=";
                for (int i = 0; i < keywords->size(); ++i)
                    lines << "-" + TagStorage::XmpWriteTag + (i == 0 ? "=" : "+=") + oneLine(keywords->at(i));
            }
            lines << file;
            return lines;
        };

        QString output, problems;
        if (!runWithArgFile(buildLines({}), 30000, output, problems, error))
            return false;

        // Some tags are unknown to ExifTool: write those again through a temporary config
        // (everything else is simply rewritten with the same values)
        const QSet<QString> custom = customXmpKeys(problems, tags);
        if (!custom.isEmpty()) {
            QStringList names;
            for (const QString& key : custom)
                names << key.section(':', -1);
            QTemporaryFile config;
            if (!writeCustomXmpConfig(config, names)) {
                error = QStringLiteral("Could not create a temporary file.");
                return false;
            }
            output.clear();
            problems.clear();
            if (!runWithArgFile(buildLines(custom), 30000, output, problems, error, config.fileName()))
                return false;
        }
        // ExifTool can exit 0 even when some tags failed to write, so also check its own
        // "N image files updated" summary line and anything printed to stderr.
        if (!problems.isEmpty() || !output.contains(QLatin1String("1 image files updated"))) {
            error = problems.isEmpty() ? output.trimmed() : problems;
            error += QStringLiteral("\n(Other changes may have been saved.)");
            return false;
        }
        return true;
    }

    bool writeTags(const QString& file, const QMap<QString, QString>& tags, QString& error)
    {
        return writeTags(file, tags, std::nullopt, error);
    }

    // Finds every tagged file in `folder` in one ExifTool call ("-if" filters to files that have
    // either tag field set), instead of reading each file individually.
    bool scanTags(const QString& folder, bool recursive, QMap<QString, QStringList>& out, QString& error)
    {
        out.clear();
        // Only files that have tags are printed. ExifTool exits with an error status when no file
        // matches, so the exit code is not checked; empty output simply means "nothing tagged".
        // (No -fast: it skips the ID3 chunk at the end of WAV files.)
        QStringList lines = { "-json", "-q", "-if",
                             "$" + TagStorage::XmpWriteTag + " or $" + TagStorage::Id3Name,
                             "-" + TagStorage::XmpWriteTag, "-" + TagStorage::Id3Name };
        if (recursive)
            lines << "-r";
        lines << folder;

        QString output, problems;
        if (!runWithArgFile(lines, 120000, output, problems, error))
            return false;
        if (output.trimmed().isEmpty())
            return true;

        QJsonParseError parseError;
        const QJsonDocument doc = QJsonDocument::fromJson(output.toUtf8(), &parseError);
        if (parseError.error != QJsonParseError::NoError || !doc.isArray()) {
            error = problems.isEmpty() ? QStringLiteral("Could not read ExifTool output.") : problems;
            return false;
        }
        for (const QJsonValue& entry : doc.array()) {
            const QJsonObject object = entry.toObject();
            // XMP tags come back as a JSON array ("Subject"); ID3 tags as one ';'-joined string
            QStringList tags = toStringList(object.value("Subject"));
            for (const QString& id3Tag : toStringList(object.value(TagStorage::Id3Name)).join(';').split(';', Qt::SkipEmptyParts))
                tags << id3Tag;
            if (!tags.isEmpty())
                out.insert(object.value("SourceFile").toString(), tags);
        }
        return true;
    }

    // Reads every supported file's full metadata in one ExifTool call, instead of spawning a
    // process per file; used by the GUI's folder-wide metadata filter.
    bool readFolderMetadata(const QString& folder, bool recursive, QMap<QString, QJsonObject>& out,
        QString& error)
    {
        out.clear();
        // No -fast: it skips the end of the file, where WAV files keep their ID3 chunk (and MP3s
        // their ID3v1 tag), so the filter would never see those tags.
        QStringList lines = { "-json", "-G" };
        if (recursive)
            lines << "-r";
        lines << folder;

        QString output, problems;
        if (!runWithArgFile(lines, 120000, output, problems, error))
            return false;

        QJsonParseError parseError;
        const QJsonDocument doc = QJsonDocument::fromJson(output.toUtf8(), &parseError);
        if (parseError.error != QJsonParseError::NoError || !doc.isArray()) {
            error = problems.isEmpty() ? QStringLiteral("Could not read ExifTool output.") : problems;
            return false;
        }
        for (const QJsonValue& entry : doc.array()) {
            const QJsonObject object = entry.toObject();
            out.insert(object.value("SourceFile").toString(), object);
        }
        return true;
    }

} // namespace ExifTool

#include "QuadMGUI/mainwindow.h"

#include <QApplication>
#include <QLocale>
#include <QTranslator>

int main(int argc, char *argv[])
{
    QApplication a(argc, argv);

    QTranslator translator;
    const QStringList uiLanguages = QLocale::system().uiLanguages();
    for (const QString &locale : uiLanguages) {
        const QString baseName = "QuadMGUI_" + QLocale(locale).name();
        if (translator.load(":/i18n/" + baseName)) {
            a.installTranslator(&translator);
            break;
        }
    }
    MainWindow w;
    w.show();
    return QApplication::exec();
}

#pragma once

#include "backend/metadata.h"
#include "backend/metadatahandler.h"

#include <QString>

#include <memory>
#include <vector>

// The one entry point the GUI uses to read and write a file's metadata.
class MetadataService
{
public:
    MetadataService();

    // Reads the file with ExifTool and hides the tags we never show (see metadatafilter.cpp).
    bool readMetadata(const QString& path, Metadata& out, QString& error) const;

    // Writes the edited fields of `data` to the file. On success the edits count as saved.
    // Fails with a readable message for file types nobody can write or for read-only tags.
    bool writeMetadata(const QString& path, Metadata& data, QString& error) const;

    // Whether the file's type can be written at all (the GUI can disable editing if not).
    bool canWrite(const QString& path) const;

    // Whether the GUI may offer an edit control for this tag.
    static bool isEditable(const QString& key);

    // Finds the files in `folder` (and subfolders if `recursive`) that carry the tag, e.g. "forest_zone_1".
    // `files` gets their full paths. This runs ExifTool over the folder, so it can take a while.
    bool findByTag(const QString& folder, const QString& tag, bool recursive, QStringList& files,
        QString& error) const;

    // Every tagged file in the folder with its tags ("path" -> tags). Useful to build a list of
    // existing tags for suggestions, or an index so searching doesn't need to scan again.
    bool scanFolderTags(const QString& folder, bool recursive, QMap<QString, QStringList>& out,
        QString& error) const;

    // Every supported file's filtered metadata fields in the folder ("path" -> fields), in one
    // ExifTool call. Used to search/filter the file list by metadata, not just tags.
    bool scanFolderMetadata(const QString& folder, bool recursive,
        QMap<QString, QMap<QString, QString>>& out, QString& error) const;

private:
    // Picks the first handler (in registration order) whose canHandle() accepts this file.
    MetadataHandler* getHandler(const QString& path) const;

    // Checked in order; the first handler that can handle the file wins.
    std::vector<std::unique_ptr<MetadataHandler>> m_handlers;
};

#include "metadataservice.h"

#include "backend/exiftoolhandler.h"
#include "backend/metadatafilter.h"
#include "backend/taglibhandler.h"
#include "exiftool.h"

#include <QJsonArray>

MetadataService::MetadataService()
{
    // Order matters: the first handler whose canHandle() returns true wins, and TagLib must
    // claim MP3/WAV before the generic ExifTool handler gets a chance to.
    m_handlers.push_back(std::make_unique<TagLibHandler>());   // MP3, WAV
    m_handlers.push_back(std::make_unique<ExifToolHandler>()); // everything else ExifTool can write
}

// Reads a file's metadata and turns it into a GUI-ready Metadata object: strips noise, adds
// blank placeholder fields, and separates out the tags (labels) from ordinary fields.
bool MetadataService::readMetadata(const QString& path, Metadata& out, QString& error) const
{
    QJsonObject raw;
    if (!ExifTool::readMetadata(path, raw, error))
        return false;

    // The tags are kept apart from the normal fields (the GUI shows them as their own rows, not
    // mixed in with the rest of the metadata table)
    QMap<QString, QString> fields = filterMetadata(raw);
    const QString fileType = raw.value(QStringLiteral("File:FileType")).toString();
    addExpectedTags(fields, fileType);
    QStringList tags;
    // XMP tags come back as either a JSON array (multiple) or a single string/number (just one)
    const QJsonValue xmp = raw.value(TagStorage::XmpReadKey);
    if (xmp.isArray()) {
        for (const QJsonValue& item : xmp.toArray())
            tags << item.toVariant().toString();
    }
    else if (xmp.isString() || xmp.isDouble()) {
        tags << xmp.toVariant().toString();
    }
    // ID3 tags are one ';'-joined string instead
    tags << raw.value(TagStorage::Id3ReadKey).toString().split(';', Qt::SkipEmptyParts);
    // Tags live in these two fields on disk, but shouldn't also show up as editable metadata rows
    fields.remove(TagStorage::XmpReadKey);
    fields.remove(TagStorage::Id3ReadKey);
    tags.removeDuplicates();

    out.load(fileType, fields, tags);
    return true;
}

// Validates and writes only the changed fields of `data`, through whichever handler claims
// this file type.
bool MetadataService::writeMetadata(const QString& path, Metadata& data, QString& error) const
{
    if (!data.hasChanges())
        return true;

    // Reject any attempt to write a read-only/composite tag before touching the file at all
    const QMap<QString, QString> changes = data.changes();
    for (auto it = changes.begin(); it != changes.end(); ++it) {
        if (!isEditable(it.key())) {
            error = QStringLiteral("%1 can't be edited.").arg(it.key());
            return false;
        }
    }

    MetadataHandler* handler = getHandler(path);
    if (!handler) {
        error = QStringLiteral("This file type can't be written.");
        return false;
    }
    if (!handler->write(path, data, error))
        return false;

    // Only mark as saved once the handler actually wrote successfully
    data.markSaved();
    return true;
}

bool MetadataService::scanFolderTags(const QString& folder, bool recursive,
    QMap<QString, QStringList>& out, QString& error) const
{
    return ExifTool::scanTags(folder, recursive, out, error);
}

// Batch-reads every file in the folder in one ExifTool call and runs each result through the
// same filterMetadata() used for single-file reads, so the filter bar sees the same fields
// as the editor would show.
bool MetadataService::scanFolderMetadata(const QString& folder, bool recursive,
    QMap<QString, QMap<QString, QString>>& out, QString& error) const
{
    out.clear();
    QMap<QString, QJsonObject> raw;
    if (!ExifTool::readFolderMetadata(folder, recursive, raw, error))
        return false;

    for (auto it = raw.begin(); it != raw.end(); ++it) {
        QMap<QString, QString> fields = filterMetadata(it.value());
        // Same blank "expected" tags the editor shows, so filtering by a tag name (e.g. "Artist")
        // also matches files of that type that don't have a value for it yet
        addExpectedTags(fields, it.value().value(QStringLiteral("File:FileType")).toString());
        out.insert(it.key(), fields);
    }
    return true;
}

bool MetadataService::findByTag(const QString& folder, const QString& tag, bool recursive,
    QStringList& files, QString& error) const
{
    files.clear();
    QMap<QString, QStringList> tagged;
    if (!ExifTool::scanTags(folder, recursive, tagged, error))
        return false;

    const QString wanted = Metadata::normalizeTag(tag);
    for (auto it = tagged.begin(); it != tagged.end(); ++it) {
        for (const QString& fileTag : it.value()) {
            // Files tagged by other programs may not be normalized, so compare normalized forms
            if (Metadata::normalizeTag(fileTag) == wanted) {
                files << it.key();
                break;
            }
        }
    }
    return true;
}

bool MetadataService::canWrite(const QString& path) const
{
    return getHandler(path) != nullptr;
}

bool MetadataService::isEditable(const QString& key)
{
    // Composite tags are calculated by ExifTool, so they can't be written either
    return !isReadOnlyTag(key) && !key.startsWith(QLatin1String("Composite:"));
}

// Returns the first registered handler that claims this file, or nullptr if none does.
MetadataHandler* MetadataService::getHandler(const QString& path) const
{
    for (const auto& handler : m_handlers) {
        if (handler->canHandle(path))
            return handler.get();
    }
    return nullptr;
}

#include "exiftoolhandler.h"

#include "exiftool.h"

#include <QFileInfo>

bool ExifToolHandler::canHandle(const QString& path) const
{
    return ExifTool::writableExtensions().contains(QFileInfo(path).suffix().toLower());
}

bool ExifToolHandler::write(const QString& path, const Metadata& data, QString& error)
{
    // The tags are stored as the file's keyword list; only rewrite it if the tags changed
    std::optional<QStringList> keywords;
    if (data.tagsChanged())
        keywords = data.tags();
    return ExifTool::writeTags(path, data.changes(), keywords, error);
}

#pragma once

#include <QMap>
#include <QString>
#include <QStringList>

// Where a file's tags (labels such as "forest_zone_1") are stored inside the file.
namespace TagStorage {
    // ExifTool formats (PNG, JPG, MP4, ...): the standard XMP keywords list.
    inline const QString XmpReadKey = QStringLiteral("XMP:Subject");     // name when reading with -G
    inline const QString XmpWriteTag = QStringLiteral("XMP-dc:Subject"); // name when writing
    // MP3 and WAV: one custom ID3v2 frame holding the tags separated by ';'.
    inline const QString Id3Name = QStringLiteral("QuadM_Tags");
    inline const QString Id3ReadKey = QStringLiteral("ID3:QuadM_Tags");
} // namespace TagStorage

// The metadata of one file: its type, a dictionary of "Group:Tag" -> text value, and a list of
// tags (labels the user attaches to find files again). It remembers what it was loaded with,
// so it can report which fields and tags were edited.
class Metadata
{
public:
    // The file type as reported by ExifTool, e.g. "MP3", "PNG" (used to pick expected tags).
    QString fileType() const { return m_fileType; }

    // Fills the object with freshly read data and forgets any edits (m_fields becomes equal to
    // m_original, so changes() starts out empty until set()/addTag()/removeTag() are called).
    void load(const QString& fileType, const QMap<QString, QString>& fields, const QStringList& tags);

    // Whether a field key (e.g. "ID3:Title") currently has a value.
    bool contains(const QString& key) const { return m_fields.contains(key); }
    // The current value of a field, or an empty string if it isn't set.
    QString get(const QString& key) const { return m_fields.value(key); }
    // Every current field, including ones that haven't changed since load().
    const QMap<QString, QString>& fields() const { return m_fields; }

    // Sets a value. An empty value means "remove this tag from the file".
    void set(const QString& key, const QString& value);

    // Only the fields that were added or changed since load()/markSaved(); this is what actually
    // gets written to disk, so unchanged fields are never rewritten.
    QMap<QString, QString> changes() const;

    // --- Tags (labels the user attaches to a file, e.g. "forest_zone_1"; shown in the GUI as rows) ---
    // The current list of tags.
    QStringList tags() const { return m_tags; }

    // Cleans a tag: trims, lower-cases, spaces become '_', anything except letters, digits,
    // '_' and '-' is removed. "Forest Zone 1" -> "forest_zone_1". Empty result = invalid.
    static QString normalizeTag(const QString& tag);

    // Adds a tag (after normalizing it). Returns false if it is invalid or already present.
    bool addTag(const QString& tag);
    // Removes a tag (compared without regard to upper/lower case).
    void removeTag(const QString& tag);
    // Whether a tag (compared without regard to upper/lower case) is currently present.
    bool hasTag(const QString& tag) const;
    // Whether the tag list differs from what was loaded (order doesn't count).
    bool tagsChanged() const;

    // True if any field or the tags were edited since load()/markSaved().
    bool hasChanges() const { return !changes().isEmpty() || tagsChanged(); }

    // Call after the changes were written to the file successfully: makes the current state the
    // new "original", so changes()/hasChanges()/tagsChanged() go back to reporting nothing changed.
    void markSaved();

private:
    QString m_fileType;
    QMap<QString, QString> m_fields;
    QMap<QString, QString> m_original;
    QStringList m_tags;
    QStringList m_originalTags;
};

#include "metadata.h"

// Stores the freshly-read data as both the current and the "original" state, so nothing counts
// as changed yet (m_fields == m_original, m_tags == m_originalTags).
void Metadata::load(const QString& fileType, const QMap<QString, QString>& fields,
    const QStringList& tags)
{
    m_fileType = fileType;
    m_fields = fields;
    m_original = fields;
    m_tags = tags;
    m_originalTags = tags;
}

// Just overwrites the in-memory value; nothing is written to the file until writeMetadata() runs.
void Metadata::set(const QString& key, const QString& value)
{
    m_fields.insert(key, value);
}

// Returns the fields that have been modified since the last load()/markSaved():
// - a brand-new key only counts if it got a non-empty value (an empty placeholder row that was
//   never filled in shouldn't trigger a write)
// - an existing key counts if its value differs from what was loaded
QMap<QString, QString> Metadata::changes() const
{
    QMap<QString, QString> result;
    for (auto it = m_fields.begin(); it != m_fields.end(); ++it) {
        if (!m_original.contains(it.key()) ? !it.value().isEmpty()
            : m_original.value(it.key()) != it.value())
            result.insert(it.key(), it.value());
    }
    return result;
}

// Trims and lower-cases the text, turns spaces into underscores, and drops anything that isn't a
// letter, digit, '_' or '-'. Keeps tags consistent no matter how the user typed them.
QString Metadata::normalizeTag(const QString& tag)
{
    QString result;
    for (const QChar c : tag.trimmed().toLower()) {
        if (c.isSpace())
            result += '_';
        else if (c.isLetterOrNumber() || c == '_' || c == '-')
            result += c;
    }
    return result;
}

// Normalizes the tag first, then rejects it if that leaves nothing or it's already present.
bool Metadata::addTag(const QString& tag)
{
    const QString clean = normalizeTag(tag);
    if (clean.isEmpty() || hasTag(clean))
        return false;
    m_tags.append(clean);
    return true;
}

// Removes every tag matching `tag` ignoring case (walks backwards so removing while iterating
// doesn't skip entries).
void Metadata::removeTag(const QString& tag)
{
    const QString wanted = tag.trimmed();
    for (int i = m_tags.size() - 1; i >= 0; --i) {
        if (m_tags[i].compare(wanted, Qt::CaseInsensitive) == 0)
            m_tags.removeAt(i);
    }
}

// Whether a tag (ignoring case) is currently present.
bool Metadata::hasTag(const QString& tag) const
{
    const QString wanted = tag.trimmed();
    for (const QString& existing : m_tags) {
        if (existing.compare(wanted, Qt::CaseInsensitive) == 0)
            return true;
    }
    return false;
}

// Whether the tag list differs from what was loaded.
bool Metadata::tagsChanged() const
{
    // The order of tags doesn't matter
    QStringList a = m_tags, b = m_originalTags;
    a.sort();
    b.sort();
    return a != b;
}

// Called once the handler has actually written the changes to disk.
void Metadata::markSaved()
{
    // Removed tags (empty value) are gone from the file, so drop them here too
    for (auto it = m_fields.begin(); it != m_fields.end();) {
        if (it.value().isEmpty())
            it = m_fields.erase(it);
        else
            ++it;
    }
    // Current state becomes the new baseline, so changes()/tagsChanged() report nothing until
    // the GUI edits something again.
    m_original = m_fields;
    m_originalTags = m_tags;
}

#include "metadatafilter.h"

#include <QJsonArray>
#include <QJsonDocument>
#include <QSet>

namespace {

    // Everything in these groups is dropped (the part before the colon in "Group:Tag").
    // ICC_Profile is deliberately NOT here: designers need the color profile.
    const QSet<QString> kIgnoredGroups = {
        "ExifTool",    // ExifTool's own version/warnings
        "System",
        "MakerNotes",  // camera-specific internals
    };

    // Individual tags that are dropped. Add new "useless" tags here.
    const QSet<QString> kIgnoredTags = {
        "SourceFile",
        "File:Directory",
        "File:FilePermissions",
        "File:FileAccessDate",     // changes every time the file is read
        "File:FileInodeChangeDate",
        "File:FileTypeExtension",
        "File:ID3Size",
        "EXIF:ThumbnailImage",
        "EXIF:PrintIM",
    };

    // Shown, but ExifTool can't write them (they are part of the PNG header or computed),
    // so the GUI should not offer edit controls. Gamma, sRGB and pixel density ARE writable.
    const QSet<QString> kReadOnlyTags = {
        "PNG:ImageWidth",
        "PNG:ImageHeight",
        "PNG:BitDepth",
        "PNG:ColorType",
        "PNG:Compression",
        "PNG:Filter",
        "PNG:Interlace",
        "File:FileName",
        "File:FileSize",
        "File:FileType",
        "File:MIMEType",
        "File:FileModifyDate",
        "File:FileCreateDate",
        "Composite:ImageSize",
        "Composite:Megapixels",
    };

    // Turns any JSON value (string, number, bool, array, object) into display text.
    QString valueToString(const QJsonValue& value)
    {
        switch (value.type()) {
        case QJsonValue::String:
            return value.toString();
        case QJsonValue::Array:
            return QString::fromUtf8(QJsonDocument(value.toArray()).toJson(QJsonDocument::Compact));
        case QJsonValue::Object:
            return QString::fromUtf8(QJsonDocument(value.toObject()).toJson(QJsonDocument::Compact));
        default:
            return value.toVariant().toString(); // numbers and booleans
        }
    }

    // ExifTool prints binary data (thumbnails, previews) as text starting with one of these.
    bool isBinaryPlaceholder(const QString& text)
    {
        return text.startsWith(QLatin1String("(Binary data")) || text.startsWith(QLatin1String("base64:"));
    }

    // Commonly-edited tags per file type, so the GUI can offer a blank row for them even
    // when the file doesn't have a value yet. "ID3:*" is handled by TagLibHandler, the rest by
    // ExifToolHandler; both accept writes to keys that aren't in the file.
    const QMap<QString, QStringList> kExpectedTagsByFileType = {
        { "MP3",{ "ID3:Title", "ID3:Artist", "ID3:Album", "ID3:Genre", "ID3:Comment", "ID3:Year", "ID3:Track" } },
        { "WAV",{ "ID3:Title", "ID3:Artist", "ID3:Album", "ID3:Genre", "ID3:Comment", "ID3:Year", "ID3:Track" } },
        { "JPEG",{ "XMP:Title", "XMP:Description", "XMP:Creator", "XMP:Rights" } },
        { "PNG",{ "XMP:Title", "XMP:Description", "XMP:Creator", "XMP:Rights" } },
        { "TIFF",{ "XMP:Title", "XMP:Description", "XMP:Creator", "XMP:Rights" } },
        { "MP4",{ "XMP:Title", "XMP:Description", "XMP:Creator", "XMP:Rights" } },
    };

} // namespace

bool isIgnoredTag(const QString& key)
{
    // A tag is ignored if it's listed individually, or its whole group is ignored
    return kIgnoredTags.contains(key) || kIgnoredGroups.contains(key.section(':', 0, 0));
}

bool isReadOnlyTag(const QString& key)
{
    return kReadOnlyTags.contains(key);
}

void addExpectedTags(QMap<QString, QString>& fields, const QString& fileType)
{
    // Only touches keys that aren't already present, so real values are never overwritten
    for (const QString& key : kExpectedTagsByFileType.value(fileType.toUpper()))
        if (!fields.contains(key))
            fields.insert(key, QString());
}

// Walks every key ExifTool returned, drops the ignored ones and binary placeholders, and
// converts every remaining JSON value to plain text.
QMap<QString, QString> filterMetadata(const QJsonObject& raw)
{
    QMap<QString, QString> result;
    for (auto it = raw.begin(); it != raw.end(); ++it) {
        if (isIgnoredTag(it.key()))
            continue;

        const QString text = valueToString(it.value());
        if (isBinaryPlaceholder(text))
            continue;

        result.insert(it.key(), text);
    }
    return result;
}

// Writes MP3 and WAV files with TagLib, because ExifTool can't write those formats.

// Keys are the ones ExifTool reports when reading: "ID3:Title", "RIFF:Artist", ...

{
public:
    // True only for .mp3/.wav; ExifTool handles reading these but can't reliably write them.
    bool canHandle(const QString& path) const override;
    // Opens the file with TagLib, applies every changed field/tag to the ID3v2 (and WAV INFO)
    // tag, then saves.
    bool write(const QString& path, const Metadata& data, QString& error) override;
};

#include "taglibhandler.h"

#include <QFile>
#include <QFileInfo>
#include <QStringList>

#include <id3v2framefactory.h>
#include <id3v2tag.h>
#include <infotag.h>
#include <mpegfile.h>
#include <textidentificationframe.h>
#include <wavfile.h>

namespace {

    // Converts a Qt string to the wide-character form TagLib expects.
    TagLib::String toTagLib(const QString& text)
    {
        return TagLib::String(text.toStdWString());
    }

    // TagLib wants a wide-character path on Windows and a byte path elsewhere.
#ifdef Q_OS_WIN
    using NativePath = std::wstring;
    NativePath nativePath(const QString& file) { return file.toStdWString(); }
    const wchar_t* cstr(const NativePath& p) { return p.c_str(); }
#else
    using NativePath = QByteArray;
    NativePath nativePath(const QString& file) { return QFile::encodeName(file); }
    const char* cstr(const NativePath& p) { return p.constData(); }
#endif

    // Sets one of the fields every tag type has. `name` must be lower case. Returns false if it isn't one.
    bool setStandardField(TagLib::Tag* tag, const QString& name, const QString& text)
    {
        if (name == "title")        tag->setTitle(toTagLib(text));
        else if (name == "artist")  tag->setArtist(toTagLib(text));
        else if (name == "album")   tag->setAlbum(toTagLib(text));
        else if (name == "genre")   tag->setGenre(toTagLib(text));
        else if (name == "comment") tag->setComment(toTagLib(text));
        else if (name == "year" || name == "recordingtime") tag->setYear(text.toUInt());
        else if (name == "track")   tag->setTrack(text.toUInt());
        else return false;
        return true;
    }

    bool isStandardName(const QString& name)
    {
        static const QStringList names = { "title", "artist", "album", "genre", "comment",
                                          "year", "recordingtime", "track" };
        return names.contains(name);
    }

    // Custom fields are ID3v2 "TXXX" frames: a name (description) plus a value.
    void setCustomField(TagLib::ID3v2::Tag* tag, const QString& name, const QString& value)
    {
        const TagLib::String description = toTagLib(name);
        // find() looks up the existing TXXX frame with this description, if any
        auto* frame = TagLib::ID3v2::UserTextIdentificationFrame::find(tag, description);
        if (value.isEmpty()) {
            // Empty value = remove the frame entirely (there's no such thing as an "empty" TXXX)
            if (frame)
                tag->removeFrame(frame, true);
            return;
        }
        if (!frame) {
            frame = new TagLib::ID3v2::UserTextIdentificationFrame(TagLib::String::UTF8);
            frame->setDescription(description);
            tag->addFrame(frame); // the tag takes ownership
        }
        frame->setText(toTagLib(value));
    }

    // Applies the edits (fields and tags) to the ID3v2 tag and (for WAV) the INFO tag. Returns false
    // and sets `error` for a key that can't be written to this kind of tag.
    bool applyChanges(TagLib::ID3v2::Tag* id3, TagLib::Tag* info, const Metadata& data, QString& error)
    {
        const QMap<QString, QString> changes = data.changes();
        for (auto it = changes.begin(); it != changes.end(); ++it) {
            // Keys may or may not have a "Group:" prefix (e.g. "ID3:Title" or just "Title");
            // default the group to "ID3" when there isn't one.
            const bool hasGroup = it.key().contains(':');
            const QString group = hasGroup ? it.key().section(':', 0, 0).toUpper() : QStringLiteral("ID3");
            const QString name = it.key().section(':', hasGroup ? 1 : 0);
            // "ID3", "ID3v2", "ID3v2_3", ... all mean the file's ID3v2 tag
            const bool isId3Group = group.startsWith(QLatin1String("ID3"));

            if (isStandardName(name.toLower())) {
                // Standard fields go on both the ID3v2 tag and, for WAV, the RIFF INFO chunk too
                setStandardField(id3, name.toLower(), it.value());
                if (info)
                    setStandardField(info, name.toLower(), it.value());
            }
            else if (isId3Group) {
                // Anything else under the ID3 group becomes a custom TXXX frame
                setCustomField(id3, name, it.value());
            }
            else {
                // A field meant for some other group (e.g. XMP) can't be written to an MP3/WAV file
                error = QStringLiteral("%1 can't be edited in this file type.").arg(it.key());
                return false;
            }
        }

        // Our tags (labels) are one custom frame; an empty list removes it
        if (data.tagsChanged())
            setCustomField(id3, TagStorage::Id3Name, data.tags().join(';'));
        return true;
    }

} // namespace

bool TagLibHandler::canHandle(const QString& path) const
{
    const QString ext = QFileInfo(path).suffix().toLower();
    return ext == "mp3" || ext == "wav";
}

bool TagLibHandler::write(const QString& path, const Metadata& data, QString& error)
{
    if (!QFileInfo(path).isWritable()) {
        error = QStringLiteral("The file does not exist or is read-only.");
        return false;
    }

    // Write text as UTF-16 so accented/non-Latin characters survive in every player
    TagLib::ID3v2::FrameFactory::instance()->setDefaultTextEncoding(TagLib::String::UTF16);

    const NativePath native = nativePath(path);
    bool saved = false;

    if (QFileInfo(path).suffix().compare("wav", Qt::CaseInsensitive) == 0) {
        TagLib::RIFF::WAV::File f(cstr(native));
        if (!f.isValid() || !f.ID3v2Tag()) {
            error = QStringLiteral("Could not read this WAV file.");
            return false;
        }
        // WAV files have both an ID3v2 tag and a separate RIFF INFO chunk; write standard
        // fields to both so the broadest range of players/explorers picks them up.
        if (!applyChanges(f.ID3v2Tag(), f.InfoTag(), data, error))
            return false;
        saved = f.save();
    }
    else {
        TagLib::MPEG::File f(cstr(native));
        if (!f.isValid()) {
            error = QStringLiteral("Could not read this MP3 file.");
            return false;
        }
        if (!applyChanges(f.ID3v2Tag(true), nullptr, data, error)) // true = create if missing
            return false;
        // ID3v2.3 is understood by more players than 2.4 (e.g. Windows Explorer)
        saved = f.save(TagLib::MPEG::File::ID3v2, TagLib::File::StripNone, TagLib::ID3v2::v3);
    }

    if (!saved)
        error = QStringLiteral("Could not save the file (is it open in another program?).");
    return saved;
}

bool FileManager::isSupportedFile(const QString &path)
{
    QFileInfo info(path);

    if (!info.isFile() || !info.isReadable())
        return false;

    QMimeDatabase mimeDatabase;
    QString mime = mimeDatabase.mimeTypeForFile(info).name();

    return mime.startsWith("image/") ||
           mime.startsWith("audio/") ||
           mime.startsWith("video/");
}

    ui->driveComboBox->clear();
    for (const QStorageInfo &drive : QStorageInfo::mountedVolumes()) {
        QString driveInfo;
        if (drive.isReady()) {
            // Display the drive's display name and root path, or just the root path if the display name is empty or generic.
            if (drive.displayName().isEmpty() || drive.displayName() == "Local Disk" || drive.displayName() == drive.rootPath()) {
                driveInfo = QString("%1").arg(drive.rootPath());
            }
            else {
                driveInfo = QString("%1 (%2)").arg(drive.displayName()).arg(drive.rootPath());
            }
            ui->driveComboBox->addItem(driveInfo, drive.rootPath());
        }
    }
}

// Load the root folders of the selected drive into the rootListWidget.
void MainWindow::loadRootFolders()
{
    currentPath.clear();
    pathHistory.clear();
    ui->rootListWidget->clear();

    QString path = ui->driveComboBox->currentData().toString(); //drive path

    // Use your home directory as the test root.
    QDir dir(path); // picks up drive path
    QFileIconProvider iconProvider;

    // List folders inside the home directory.
    QFileInfoList folders = dir.entryInfoList(
        QDir::Dirs | QDir::NoDotAndDotDot,
        QDir::Name
    );

    // Populate the rootListWidget with the folders.
    for (const QFileInfo& folder : folders) {
        auto* item = new QListWidgetItem(iconProvider.icon(folder), folder.fileName());

        // Store the actual directory path.
        item->setData(Qt::UserRole, folder.absoluteFilePath());

        ui->rootListWidget->addItem(item);
    }
}

// Load the contents of a directory into the contentsListWidget.
void MainWindow::loadDirectory(const QString& path)
{
    QDir dir(path);

    if (!dir.exists()) {
        qDebug() << "Directory does not exist:" << path;
        return;
    }

    QString absolutePath = path;

    // Save the current directory for Back navigation.
    if (!currentPath.isEmpty() &&QDir::cleanPath(currentPath) != QDir::cleanPath(absolutePath)) {
        pathHistory.append(currentPath);
    }

    currentPath = absolutePath;

    ui->welcomeWidget->hide();

    ui->contentsListWidget->clear();
    ui->currentPathLabel->setText(currentPath);

    ui->contentsListWidget->show();
    ui->currentPathLabel->show();
    ui->backButton->setVisible(!pathHistory.isEmpty());
    ui->filePathSearchBar->show();
    ui->tagFilterBar->show();

    QFileIconProvider iconProvider;

    // Show folders first, then supported media files.
    QFileInfoList entries = dir.entryInfoList(QDir::Dirs | QDir::Files | QDir::NoDotAndDotDot, QDir::DirsFirst | QDir::Name);

    for (const QFileInfo& entry : entries) {

        // Always allow folders so users can navigate through them.
        // Only display files accepted by FileManager.
        if (entry.isFile() && !FileManager::isSupportedFile(entry.absoluteFilePath())) {
            continue;
        }

        auto* item = new QListWidgetItem(iconProvider.icon(entry), entry.fileName());

        // Keep the full path attached to the list item.
        item->setData(Qt::UserRole, entry.absoluteFilePath());

        ui->contentsListWidget->addItem(item);
    }

    // Metadata (including tags) is scanned lazily in ensureDirectoryMetadataScanned(), only once
    // the user actually types a filter, so simply navigating folders never touches ExifTool.
    directoryMetadata.clear();
    directoryMetadataScanned = false;
    applyTagFilter(ui->tagFilterBar->text());

    //qDebug() << "Loaded directory:" << currentPath<< "| Displayed items:" << ui->contentsListWidget->count();
}


void MainWindow::ensureDirectoryMetadataScanned()
{
    if (directoryMetadataScanned)
        return;
    directoryMetadataScanned = true;

    QString error;
    service.scanFolderMetadata(currentPath, false, directoryMetadata, error);
}

void MainWindow::applyTagFilter(const QString& query)
{
    const QString wanted = query.trimmed();
    if (!wanted.isEmpty())
        ensureDirectoryMetadataScanned();

    for (int i = 0; i < ui->contentsListWidget->count(); ++i) {
        QListWidgetItem* item = ui->contentsListWidget->item(i);
        const QString itemPath = item->data(Qt::UserRole).toString();

        if (wanted.isEmpty() || QFileInfo(itemPath).isDir()) {
            item->setHidden(false);
            continue;
        }

        bool matches = false;
        const QMap<QString, QString>& fields = directoryMetadata.value(itemPath);
        for (auto it = fields.constBegin(); it != fields.constEnd(); ++it) {
            // Match the field's name (e.g. "FileName") or its value (tags live here too, as
            // XMP:Subject/ID3:QuadM_Tags).
            if (it.key().section(':', -1).contains(wanted, Qt::CaseInsensitive)
                || it.value().contains(wanted, Qt::CaseInsensitive)) {
                matches = true;
                break;
            }
        }
        item->setHidden(!matches);
    }
}

// Format metadata keys for display by inserting spaces before uppercase letters.
QString MainWindow::formatMetadataKey(const QString& key)
{
    QString result;

    for (int i = 0; i < key.length(); ++i) {
        QChar current = key[i];
        if (i > 0 && current.isUpper() && key[i - 1].isLower()) {
            result += ' ';
        }
        result += current;
    }

    return result;
}

// Populate the metadata panel with information about the selected file.
void MainWindow::metadataPopulate(const QString& filePath)
{
    QFileInfo fileInfo(filePath);

    // Stop the OIIA placeholder animation
    QMovie* catMovie = ui->metadataPlaceholderLabel->movie();

    if (catMovie) {
        catMovie->stop();
        ui->metadataPlaceholderLabel->setMovie(nullptr);
    }

    if (!fileInfo.exists() || !fileInfo.isFile()) {
        ui->metadataPlaceholderLabel->setText("File not found.");
        return;
    }

    if (!FileManager::isSupportedFile(filePath)) {
        ui->metadataPlaceholderLabel->setText(
            "This file format is not supported."
        );
        return;
    }

    QString html;

    // Basic file information
    html += "<h3>File Information</h3>";

    html += "<b>Name:</b> " + fileInfo.fileName().toHtmlEscaped() + "<br>";
    html += "<b>Type:</b> " + fileInfo.suffix().toUpper() + "<br>";
    html += "<b>Size:</b> " + QString::number(fileInfo.size() / 1024.0, 'f', 2) + " KB<br>";

    html += "<b>Modified:</b> " + fileInfo.lastModified().toString("yyyy-MM-dd hh:mm:ss") + "<br>";

    html += "<hr>";

    // File path
    html += "<h3>File Path</h3>";
    html += fileInfo.absoluteFilePath().toHtmlEscaped();

    html += "<hr>";

    // Extract media metadata
    QJsonObject metadata;
    QString error;

    bool success = ExifTool::readMetadata(
        fileInfo.absoluteFilePath(), metadata, error
    );

    QMap<QString, QMap<QString, QString>> groupedMetadata; //map to hold the categories and their fields

    html += "<h3>Media Metadata</h3>";

    if (success && !metadata.isEmpty()) {

        for (auto it = metadata.constBegin();it != metadata.constEnd(); ++it) {
            QStringList parts = it.key().split(':'); // split category
            if (parts.size() > 1) {
                groupedMetadata[parts[0]][parts[1]] = it.value().toVariant().toString().toHtmlEscaped(); // add the key value pair with category
                //qDebug() << groupedMetadata[parts[0]]<< " "<<groupedMetadata[parts[0]][parts[1]];
            }
            else {
                QString key = it.key().toHtmlEscaped();
                QString value = it.value().toVariant().toString().toHtmlEscaped();
                html += "<b>" + key + ":</b> " + value + "<br>";
            }
        }

        for (auto ctg = groupedMetadata.constBegin(); ctg != groupedMetadata.constEnd(); ++ctg) {
            html += "<h3>" + formatMetadataKey(ctg.key()).toHtmlEscaped() + "</h3>";

            for (auto p0 = ctg.value().constBegin(); p0 != ctg.value().constEnd(); ++p0) {
                html += "<b>" + formatMetadataKey(p0.key()).toHtmlEscaped() + ":</b> " + p0.value().toHtmlEscaped() + "<br>";
            }
        }

    }
    else if (success) {
        html += "No metadata found.";
    }
    else {
        html += "Could not extract metadata.<br>";
        html += error.toHtmlEscaped();

        //qDebug() << "ExifTool metadata error:" << error;
    }

    ui->metadataPlaceholderLabel->setAlignment(Qt::AlignLeft | Qt::AlignTop);
    
    ui->metadataPlaceholderLabel->setText(html);
    ui->metadataPlaceholderLabel->setTextFormat(Qt::RichText);
    ui->metadataPlaceholderLabel->setWordWrap(true);

    ui->metadataPlaceholderLabel->setText(html);
    ui->metadataPlaceholderLabel->setTextFormat(Qt::RichText);
    ui->metadataPlaceholderLabel->setWordWrap(true);
    ui->metadataPlaceholderLabel->setTextInteractionFlags(
        Qt::TextSelectableByMouse
    );

    //qDebug() << "Metadata displayed for:" << filePath;
}


// Open the metadata editor for the specified file path.
void MainWindow::openMetadataEditor(const QString& filePath)
{
    if (!metadataEditor) {
        metadataEditor = new MetadataEditor(filePath, this);
        // Ensure the the panels are updated when the user saves metadata in the editor.
        connect(metadataEditor, &MetadataEditor::metadataSaved, this, [this](const QString& savedPath) {
            qDebug() << "Metadata saved:" << savedPath;
            metadataPopulate(savedPath);
            loadDirectory(currentPath);
            });
    }
    else {
        metadataEditor->setPath(filePath);
    }

    metadataEditor->show();
    metadataEditor->raise();
    metadataEditor->activateWindow();
}


// Open the settings window and connect its save signal to update the theme and reload drives.
void MainWindow::handleOpenWindow() {

    Settings *subWindow = new Settings(); 

    connect(subWindow, &Settings::settingsSave, this, [this]() {
        loadDefaultDrive();
        loadRootFolders();
        QSettings settings("QuadM", "QUADM");

        QString theme =settings.value("theme", "Dark").toString();
    
        applyTheme(theme);
    });
    
    subWindow->setAttribute(Qt::WA_DeleteOnClose); 
    subWindow->show();
}

// Apply the selected theme by loading the corresponding CSS file and setting it as the application's stylesheet.
void MainWindow::applyTheme(const QString& theme)
{
    QString cssPath = QDir::currentPath() + "/src/QuadMGUI/" + theme.toLower() + ".css";

    QFile file(cssPath); // Declare the QFile object

    if (file.open(QFile::ReadOnly | QFile::Text)) {
        QString css = QString::fromUtf8(file.readAll());
        qApp->setStyleSheet(css);
        //qDebug() << "Applied theme:" << theme;
        file.close();
    }
    else {
        qDebug() << "Could not load theme:" << cssPath;
        qDebug() << file.errorString();
    }
}

        // Wait for a pause in typing before (possibly) scanning metadata and filtering,
        // so holding down a key doesn't trigger it on every single character.
        filterDebounceTimer.start();
        });

    // Clicking a root folder loads its contents.
    connect(ui->rootListWidget, &QListWidget::itemClicked, this, [this](QListWidgetItem* item) {

        //reset the path history and current path when a root folder is clicked to avoid confusion in back navigation
        pathHistory.clear();
        currentPath.clear();
       

        loadDirectory(item->data(Qt::UserRole).toString());
        });

    // Single-click a supported file to display its metadata.
    connect(ui->contentsListWidget, &QListWidget::itemClicked, this, [this](QListWidgetItem* item) {
        QString path = item->data(Qt::UserRole).toString();
        QFileInfo info(path);

        if (info.isFile() && FileManager::isSupportedFile(path)) {
            filePath = info.absoluteFilePath();

            qDebug() << "File path:" << filePath;

            // Read metadata when the user selects a supported file.
            metadataPopulate(filePath);
        }
        });

        QString searchPath = ui->filePathSearchBar->text().trimmed();

            return;

        if (url.isLocalFile()) {
            searchPath = url.toLocalFile();
        }
        QFileInfo info(searchPath);

        if (!info.exists()) {
            qDebug() << "Path does not exist:" << searchPath;
            return;
        }

        if (info.isDir()) {
            loadDirectory(info.absoluteFilePath());

        }
        else if (info.isFile() && FileManager::isSupportedFile(searchPath)) {

            filePath = info.absoluteFilePath();

            // Load the parent directory.
            loadDirectory(info.absolutePath());

            QFileIconProvider iconProvider;

            item->setData(Qt::UserRole, filePath);


            ui->contentsListWidget->addItem(item);
            ui->contentsListWidget->setCurrentItem(item);

            // Populate metadata.
            metadataPopulate(filePath);
        }
        else {
            qDebug() << "Unsupported file type or invalid path:" << searchPath;
        }
        });

    ui->contentsListWidget->setContextMenuPolicy(Qt::CustomContextMenu);

    connect(ui->contentsListWidget, &QListWidget::customContextMenuRequested, this, [this](const QPoint& pos) {

        QListWidgetItem* item = ui->contentsListWidget->itemAt(pos);

        if (!item)
            return;

        QMenu menu(this);
        QAction* copyPathAction = menu.addAction("Copy File Path");

        QAction* selectedAction = menu.exec(ui->contentsListWidget->mapToGlobal(pos));

        if (selectedAction == copyPathAction) {
            QString filePath = item->data(Qt::UserRole).toString();
            QApplication::clipboard()->setText(filePath);
        }
        });

    // Folder logic. Double click to display its content
    connect(ui->contentsListWidget, &QListWidget::itemDoubleClicked, this, [this](QListWidgetItem* item) {
        QString path = item->data(Qt::UserRole).toString();
        QFileInfo info(path);

        if (info.isDir()) {
            loadDirectory(path);
        }
        });

    connect(ui->expandMetadataButton, &QPushButton::clicked, this, [this]() {
        if (!filePath.isEmpty()) {
            openMetadataEditor(filePath);
        }
        });

    //Navigate backward
    connect(ui->backButton, &QPushButton::clicked, this, [this]() {

        if (pathHistory.isEmpty())
            return;
    
        QString previousPath = pathHistory.takeLast();
    
        // Prevent loadDirectory() from adding the current path again.
        currentPath.clear();
    
        loadDirectory(previousPath);
    
    });
}

}

void MetadataEditor::populateData()
{
    //QJsonObject metadata;
    QString error;
    QFileInfo fileInfo(path);

    if (!fileInfo.exists() || !fileInfo.isFile()) {
        ui->fileNameLabel->setText("File not found.");
        return;
    }

    ui->fileNameLabel->setText(fileInfo.fileName());
    ui->fileTypeLabel->setText(fileInfo.suffix().toUpper());

    // Clear existing widgets
    while (QLayoutItem *item = ui->metadataLayout->takeAt(0)) {
        if (QWidget *widget = item->widget())
            widget->deleteLater();
        delete item;
    }

    metadataFields.clear();
    metadataRows.clear();
    originalMetadata.clear();
    deletedMetadata.clear();

    if (!service.readMetadata(path, metadata, error)) {
        ui->metadataLayout->addWidget(new QLabel(error));
        return;
    }

    //TEMPORARY DEBUG
    // qDebug() << "Fields returned by service:" << metadata.fields().size();

    // for (auto it = metadata.fields().constBegin();
    //     it != metadata.fields().constEnd(); ++it) {
    //     qDebug() << it.key() << "=" << it.value();
    // }

    // Store original metadata, including empty and ignored fields
    for (auto it = metadata.fields().constBegin();
         it != metadata.fields().constEnd(); ++it) {
        originalMetadata[it.key()] = it.value();
    }

    // Populate metadata fields
    QSet<QString> categories;

    for (auto it = originalMetadata.constBegin();
         it != originalMetadata.constEnd(); ++it) {

        QString category = it.key().contains(':')
            ? it.key().section(':', 0, 0)
            : "Other";

        if (!categories.contains(category)) {
            auto *categoryLabel = new QLabel(category);
            categoryLabel->setStyleSheet(
                "font-size: 16px; font-weight: bold;"
            );

            ui->metadataLayout->addWidget(categoryLabel);
            categories.insert(category);
        }

        // Display every field, including ignored and empty fields
        addMetadataField(it.key(), it.value());
    }

    ui->metadataLayout->addStretch();
}



void MetadataEditor::addMetadataField(const QString &key, const QString &value)
{
    QWidget *row = new QWidget(ui->metadataScrollContents);
    QHBoxLayout *rowLayout = new QHBoxLayout(row);

    QLabel *propertyLabel = new QLabel(key.section(':', -1), row);
    QLineEdit *valueEdit = new QLineEdit(value, row);
    QPushButton *deleteButton = new QPushButton("Delete", row);
    
    const bool readOnly =  isReadOnlyTag(key) && key != "File:FileName";



    if (readOnly) {
        valueEdit->setReadOnly(true);

        //qDebug() << "Read-only metadata field:" << key;
        valueEdit->setStyleSheet("QLineEdit {"
            "background-color: #e0e0e0;"
            "color: #777777;"
            "border: 1px solid #c0c0c0;""}"
        );

        deleteButton->setEnabled(false);
        deleteButton->setToolTip("This metadata field is read-only.");
    }

    valueEdit->setMinimumWidth(300);

    rowLayout->addWidget(propertyLabel);
    rowLayout->addWidget(valueEdit);
    rowLayout->addWidget(deleteButton);

    metadataFields[key] = valueEdit;
    metadataRows[key] = row;

    ui->metadataLayout->addWidget(row); // label line

    // delte only if not read-only
    if (!readOnly) {
        connect(deleteButton, &QPushButton::clicked, this, [this, key]() {
            deleteMetadataField(key);
        });
    }
}

void MetadataEditor::addMetadata()
{
    bool ok;

    QString key = QInputDialog::getText(
        this, "Add Metadata",
        "Tag name (e.g. XMP:Description):",
        QLineEdit::Normal, "", &ok
    ).trimmed();

    if (!ok || key.isEmpty())
        return;

    if (metadataFields.contains(key)) {
        QMessageBox::warning(this, "Duplicate Metadata","This metadata tag already exists.");
        return;
    }

    if (key.startsWith('-') || key.contains('=') ||key.contains('\n') || key.contains('\r')) {
        QMessageBox::warning(this, "Invalid Tag","Please enter a valid metadata tag name.");
        return;
    }

    // Create a temporary copy
    QFileInfo info(path);
    QString tempPath = info.dir().filePath(
        info.completeBaseName() + "_temp." + info.suffix()
    );

    if (!QFile::copy(path, tempPath)) {
        QMessageBox::critical(this, "Validation Error","Could not create a temporary file for validation.");
        return;
    }

    // Attempt a dummy write
    QMap<QString, QString> testTags;
    testTags.insert(key, "QuadM_Test");

    QString error;
    bool valid = ExifTool::writeTags(tempPath, testTags, error);

    // Delete temporary copy
    QFile::remove(tempPath);

    if (!valid) {
        QMessageBox::warning(this, "Invalid Metadata",error);
        return;
    }

    // Collect actual value
    QString value = QInputDialog::getText(this, "Add Metadata", "Value:",QLineEdit::Normal, "", &ok);

    if (!ok)
        return;

    deletedMetadata.remove(key);
    addMetadataField(key, value);
}

void MetadataEditor::deleteMetadataField(const QString &key)
{
    QWidget *row = metadataRows.value(key, nullptr);

    if (!row)
        return;

    if (originalMetadata.contains(key))
        deletedMetadata.insert(key);

    ui->metadataLayout->removeWidget(row);
    metadataFields.remove(key);
    metadataRows.remove(key);

    row->deleteLater();
}

void MetadataEditor::saveMetadata()
{
   //QMap<QString, QString> changedTags;

    for (auto it = metadataFields.constBegin(); it != metadataFields.constEnd(); ++it) {
        if (it.key() == "File:FileName")
            continue;

        metadata.set(it.key(), it.value()->text());
    }

    QString newName = metadataFields.value("File:FileName")
                          ? metadataFields.value("File:FileName")->text().trimmed()
                          : QFileInfo(path).fileName();

    const bool nameChanged = newName != QFileInfo(path).fileName();

    if (!metadata.hasChanges() && deletedMetadata.isEmpty() && !nameChanged) {
        QMessageBox::information(this, "Metadata Editor", "No changes to save.");
        return;
    }

    if (!deletedMetadata.isEmpty()) {
        QMessageBox::warning(this, "Deletion Not Supported", "Deleted fields have been removed from the editor, but your current ExifTool writeTags() function does not support deleting tags from the file.");
    }

    if (!metadata.hasChanges() && !nameChanged) {
        QMessageBox::information(this, "Metadata Editor", "There are no additions or edits to save.");
        return;
    }

    // Rename file if filename changed
    if (nameChanged) {
        QFileInfo info(path);
        QString newPath = info.dir().filePath(newName);

        if (newName.isEmpty() || newName == "." || newName == ".." ||
            QFileInfo(newName).fileName() != newName || QFileInfo::exists(newPath)) {
            QMessageBox::warning(this, "Rename Error",
                                 "Invalid filename or a file with that name already exists.");
            return;
        }

        if (!QFile::rename(path, newPath)) {
            QMessageBox::warning(this, "Rename Error", "Could not rename the file.");
            return;
        }

        path = newPath;
    }

    QString error;

    if (metadata.hasChanges()) {
        if (!service.writeMetadata(path, metadata, error)) {
            QMessageBox::critical(this, "Save Error", error);
            return;
        }
    }

    QMessageBox::information(this, "Metadata Editor", "Metadata changes saved successfully.");

    populateData();
    emit metadataSaved(path);
}

        QString driveInfo;
        if (drive.isReady()) {
            if (drive.displayName().isEmpty() || drive.displayName() == "Local Disk" || drive.displayName() == drive.rootPath()) {
                driveInfo = QString("%1").arg(drive.rootPath());
            }
            else {
                driveInfo = QString("%1 (%2)").arg(drive.displayName()).arg(drive.rootPath());
            }
            ui->defaultDriveComboBox->addItem(driveInfo, drive.rootPath());
        }
    }
}

void Settings::loadSettings()
{
    QSettings settings("QUADM", "QUADM");

    QString defaultDrive =
        settings.value("defaultDrive", QDir::rootPath()).toString();

    int index = ui->defaultDriveComboBox->findData(defaultDrive);

    if (index != -1)
        ui->defaultDriveComboBox->setCurrentIndex(index);

    QString theme =settings.value("theme", "Dark").toString();

    int themeIndex = ui->themeComboBox->findData(theme);

    if (themeIndex != -1)
        ui->themeComboBox->setCurrentIndex(themeIndex);
}

void Settings::saveSettings()
{
    QSettings settings("QUADM", "QUADM");

    QString selectedDrive = ui->defaultDriveComboBox->currentData().toString();

    
    QString selectedTheme = ui->themeComboBox->currentData().toString();

    settings.setValue("defaultDrive", selectedDrive);
    settings.setValue("theme", selectedTheme);
    settings.sync();

    close();
    emit settingsSave();
}
